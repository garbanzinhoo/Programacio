public class DBViewer {

    private static final int CHAR_WIDTH = 10;

    /**
     * Crear llista amb les taules de la base de dades i afegirla al panell esquerre de la finestra.
     * @param dbName Nom de la base de dades (s'afegeix al títol de la finestra)
     * @param taules Noms de les taules (s'afegeixen a la llista)
     */
    public static void mostrarTaulesBdGUI(String dbName, String[] taules) {
        Finestra.getFinestra().setTitle("JDBC - " + dbName);

        JList l = Finestra.getLlista();
        DefaultListModel lm = (DefaultListModel) l.getModel();
        lm.removeAllElements();
        lm.addAll(Arrays.asList(taules));

        JSplitPane sp = Finestra.getSplit();
        int dl = sp.getDividerLocation();

        JScrollPane jspl = new JScrollPane(l);
        jspl.setMinimumSize(new Dimension(100, 50));
        sp.setLeftComponent(jspl);
        sp.setRightComponent(new JScrollPane());
        sp.setDividerLocation(dl);

        l.setSelectedIndex(0);
    }

    /**
     * Mostrar les taules de la base de dades.
     * @param dbName Nom de la base de dades (s'afegeix al títol de la finestra)
     * @param taules Noms de les taules (s'afegeixen a la llista)
     */
    public static void mostrarTaulesBd(String dbName, String[] taules) {
        System.out.print("\nTaules " + dbName);

        for (int i = 0; i < taules.length; ++i) {
            System.out.print("\n" + taules[i]);
        }
        System.out.println();
    }

    /**
     * Crear una JTable amb el resultat d'una consulta i afegirla al panell dret de la finestra.
     * @param dades Dades dels registres obtinguts en una consulta
     * @param camps Noms dels camps de la consulta
     * @param amplades Amplada en caràcters de cada camp
     */
    public static void mostrarTaulaGUI(Object[][] dades, Object[] camps, int[] amplades) {
        // TODO: seria millor crear un AbstractTableModel
        JTable taula = new JTable(dades, camps);
        taula.setAutoResizeMode(JTable.AUTO_RESIZE_OFF);
        for (int i = 0; i < amplades.length; ++i) {
            taula.getColumnModel().getColumn(i).setPreferredWidth(amplades[i] * CHAR_WIDTH);
        }

        JSplitPane sp = Finestra.getSplit();
        int dl = sp.getDividerLocation();

        JScrollPane jspr = new JScrollPane(taula);
        jspr.setMinimumSize(new Dimension(100, 50));
        sp.setRightComponent(jspr);
        sp.setDividerLocation(dl);
    }

    /**
     * Mostrar per consola el resultat d'una consulta.
     * @param dades Dades dels registres obtinguts en una consulta
     * @param camps Noms dels camps de la consulta
     * @param amplades Amplada en caràcters de cada camp
     */
    public static void mostrarTaula(Object[][] dades, Object[] camps, int[] amplades) {
        int c = camps.length;

        System.out.print("\n ");
        for (int j = 0; j < c; ++j) {
            System.out.printf("%-" + amplades[j] + "s ", camps[j]);
        }
        System.out.print("\n+");
        for (int j = 0; j < c; ++j) {
            System.out.printf("%-" + amplades[j] + "s+", new String(new char[amplades[j]]).replace("\0", "-"));
        }
        for (Object[] d : dades) {
            System.out.print("\n ");
            for (int j = 0; j < c; ++j) {
                System.out.printf("%-" + amplades[j] + "s ", d[j] == null ? "" : d[j]);
            }
        }
        System.out.println();
    }



    /**
     * @param args the command line arguments
     */
    public static void main(String[] args) {
        Finestra.crearFinestra();
    }
}

public class Control {
    // Controlador per detectar que s'ha seleccionat una altra taula
    public static ListSelectionListener listControl = new ListSelectionListener() {
        @Override
        public void valueChanged(ListSelectionEvent e) {
            if (!e.getValueIsAdjusting()) {
                JList list = (JList)e.getSource();
                Model.obrirTaula((String)list.getSelectedValue(), Finestra.propertiesSelected());
                DBViewer.mostrarTaulaGUI(Model.getDades(), Model.getCamps(), Model.getAmplades());
                DBViewer.mostrarTaula(Model.getDades(), Model.getCamps(), Model.getAmplades());
            }
        }
    };

    // Controlador per gestionar el menú i la barra d'eines
    public static ActionListener buttonControl = new ActionListener() {
        @Override
        public void actionPerformed(ActionEvent e) {
            String ac = e.getActionCommand();
            File file;

            switch(ac) {
                case "openDB":
                    file = Dialegs.getFileToOpen("Data Base Files", "db");
                    if (file == null) return;
                    if (!Files.exists(Path.of(file.getPath()))) {
                        Dialegs.showMessage("L'arxiu no existeix");
                        return;
                    }

                    Finestra.activarBotons(true);
                    carregarBD(file.getAbsolutePath());
                    break;
                case "data/prop":
                    Finestra.selectProperties(!Finestra.propertiesSelected());
                    carregarTaula();
                    break;
                case "executeFile":
                    file = Dialegs.getFileToOpen("SQL Files", "sql");
                    if (file == null) return;
                    if (!Files.exists(Path.of(file.getPath()))) {
                        Dialegs.showMessage("L'arxiu no existeix");
                        return;
                    }

                    execute(readFile(file));
                    break;
                case "executeText":
                    execute("");
                    break;
            }
        }

        private void carregarBD(String path) {
            Model.obrirDB(path);
            DBViewer.mostrarTaulesBd(Model.getDbName(), Model.getTaules());
            DBViewer.mostrarTaulesBdGUI(Model.getDbName(), Model.getTaules());
        }

        private void carregarTaula() {
            Model.obrirTaula(Finestra.getSelectedTable(), Finestra.propertiesSelected());
            DBViewer.mostrarTaula(Model.getDades(), Model.getCamps(), Model.getAmplades());
            DBViewer.mostrarTaulaGUI(Model.getDades(), Model.getCamps(), Model.getAmplades());
        }

        private String readFile(File f) {
            String t = "";

            try {
                Scanner reader = new Scanner(f);

                while (reader.hasNext()) {
                    t += reader.nextLine() + "\n";
                }

                reader.close();
            } catch (Exception e) {
                e.printStackTrace();
            }

            return t;
        }

        private void execute(String sql) {
            sql = Dialegs.execSQL(sql);
            if (sql == null) return;

            boolean actDB = false;
            boolean actTable = false;
            String command;
            sql = sql.replaceAll("\\s+", " ").trim();

            String[] sentencies = sql.split("\\s*;\\s*(?=([^'\"]*['\"][^'\"]*['\"])*[^'\"]*$)");
            for (int i = 0; i < sentencies.length; ++i) {
                command = sentencies[i].split(" ", 2)[0].toUpperCase();
                System.out.println(sentencies[i]);
                switch(command) {
                    case "SELECT":
                        // TODO: mostrar resultat per consola o finestra/pestanyes/diàleg?
                        break;
                    case "CREATE":
                    case "ALTER":
                    case "DROP":
                        Model.actualitzar(sql);
                        actDB = true;
                        break;
                    case "INSERT":
                    case "UPDATE":
                    case "DELETE":
                        Model.actualitzar(sql);
                        actTable = true;
                        break;
                    default:
                        Model.executar(sql);
                        // TODO: carregar BD o taula?
                }
            }

            if (actDB) {
                carregarBD(Model.getDbName());
            } else if (actTable) {
                carregarTaula();
            }
        }
    };
}

public class Model {

    // Nom de la DB
    private static String dbName;
    // Noms de les taules de la BD
    private static String[] taules;
    // Camps de la taula (nom i tipus)
    private static String[] camps;
    // Dades de la taula
    private static String[][] dades;
    // Amplada màxima en caràcters de cada camp per adaptar l'amplada de les columnes de la taula
    private static int[] amplades;
    // Consulta per obtenir els noms de les taules de la BD
    private static final String qGetTables = "SELECT name FROM sqlite_schema WHERE type='table' ORDER BY name";



    public static String getDbName() {
        return dbName;
    }

    public static String[] getTaules() {
        return taules;
    }

    public static String[] getCamps() {
        return camps;
    }

    public static String[][] getDades() {
        return dades;
    }

    public static int[] getAmplades() {
        return amplades;
    }

    /**
     * Guardar les taules de la base de dades incloent 'sqlite_schema'.
     * @param db Ruta a la BD
     */
    public static void obrirDB(String db) {
        Connection con = JDBC.openCon("jdbc:sqlite:" + db);
        if (con == null) return;

        dbName = db;

        try {
            Statement stmt = con.createStatement();
            ResultSet rs = stmt.executeQuery(qGetTables);
            String[][] data = JDBC.getRsValues(rs);

            taules = new String[data.length + 1];
            taules[0] = "sqlite_schema";
            for (int i = 1; i < taules.length; ++i) {
                taules[i] = data[i - 1][0];
            }
        } catch (SQLException e) {
            System.out.println(e.getMessage());
        } finally {
            JDBC.closeCon(con);
        }
    }

    /**
     * Obtenir i guardar les dades o la informació dels camps d'una taula.
     * @param tableName Nom de la taula
     * @param info Si és 'true', guarda la informació dels camps
     */
    public static void obrirTaula(String tableName, boolean info) {
        Connection con = JDBC.openCon("jdbc:sqlite:" + dbName);
        if (con == null) return;

        try {
            Statement stmt = con.createStatement();
            String sql = "SELECT * FROM " + tableName;
            ResultSet rs = stmt.executeQuery(sql);
            String[][] fieldsData;
            if (info) {
                fieldsData = new String[][] {{"camp", ""}, {"tipus", ""}};
                dades = JDBC.getRsFields(rs);
            } else {
                fieldsData = JDBC.getRsFields(rs);
                dades = JDBC.getRsValues(rs);
            }
            if (fieldsData != null && dades != null) {
                camps = new String[fieldsData.length];
                for (int i = 0; i < camps.length; ++i) {
                    camps[i] = fieldsData[i][0];
                }

                amplades = JDBC.getMaxChars(dades, camps, true);
            }
        } catch (SQLException e) {
            System.out.println(e.getMessage());
        } finally {
            JDBC.closeCon(con);
        }
    }

    public static int actualitzar(String query) {
        return JDBC.update(dbName, query);
    }

    public static void consultar(String query) {
        ResultSet rs = JDBC.select(dbName, query);
    }

    public static void executar(String query) {
        JDBC.execute(dbName, query);
    }

}

Métodes genèrics per accedir a bases de dades

public class JDBC {

    /**
     * Connectar amb una BD indicada en el paràmetre 'url'.
     * @param url Protocol i ruta cap a la BD<br>Exemple: "dbviewer:sqlite:prova.db"
     * @return Connexió amb la BD o 'null' si no s'ha pogut connectar
     */
    public static Connection openCon(String url) {
        Connection con = null;

        try {
            con = DriverManager.getConnection(url);
        } catch (SQLException e) {
            System.out.println(e.getMessage());
        }

        return con;
    }

    /**
     * Desconnectar d'una BD.
     * @param con Connexió amb la BD
     */
    public static void closeCon(Connection con) {
        try {
            if (con != null) con.close();
        } catch (SQLException e) {
            System.out.println(e.getMessage());
        }
    }

    /**
     * Fer una sentència que modifica la base de dades.
     * @param db Path de la base de dades
     * @param query Sentència SQL
     * @return Valor que retorna executeUpdate(query)
     */
    public static int update(String db, String query) {
        Connection con = JDBC.openCon("jdbc:sqlite:" + db);
        if (con == null) return -1;

        int r = -1;

        try {
            Statement stmt = con.createStatement();
            r = stmt.executeUpdate(query);
            System.out.println(r);
        } catch (SQLException e) {
            System.out.println(e.getMessage());
        } finally {
            JDBC.closeCon(con);
        }

        return r;
    }

    /**
     * Fer una sentència de consulta a la base de dades.
     * @param db Path de la base de dades
     * @param query Sentència SQL
     * @return ResultSet que retorna executeQuery(query)
     */
    public static ResultSet select(String db, String query) {
        Connection con = JDBC.openCon("jdbc:sqlite:" + db);
        if (con == null) return null;

        ResultSet rs = null;

        try {
            Statement stmt = con.createStatement();
            rs = stmt.executeQuery(query);
        } catch (SQLException e) {
            System.out.println(e.getMessage());
        } finally {
            JDBC.closeCon(con);
        }

        return rs;
    }

    /**
     * Fer una sentència o sentències SQL genèriques.
     * @param db Path de la base de dades
     * @param query Sentència SQL
     */
    public static void execute(String db, String query) {
        Connection con = JDBC.openCon("jdbc:sqlite:" + db);
        if (con == null) return;

        try {
            Statement stmt = con.createStatement();
            stmt.execute(query);
        } catch (SQLException e) {
            System.out.println(e.getMessage());
        } finally {
            JDBC.closeCon(con);
        }
    }

    /**
     * Obtenir els valors de tots els camps de cada registre de la consulta
     * @param rs Conjunt de registres retornats per la consulta
     * @return Array amb els valors en forma d'String
     */
    public static String[][] getRsValues(ResultSet rs) {
        LinkedList<String[]> regList = new LinkedList<>();
        int nRows = 0;
        int nCols = 0;
        int i;

        try {
            ResultSetMetaData rsmd = rs.getMetaData();
            nCols = rsmd.getColumnCount();

            while(rs.next()) {
                String[] s = new String[nCols];
                for (i = 0; i < nCols; i++) {
                    // TODO: obtenir valor segons el tipus de dada (int, float...) ???
                    s[i] =  rs.getString(i + 1);
                }
                regList.add(s);
                nRows++;
            }
        } catch (SQLException throwables) {
            throwables.printStackTrace();
        }

        i = 0;
        String[][] registres = new String[nRows][nCols];
        while (!regList.isEmpty()) {
            registres[i] = regList.remove();
            ++i;
        }

        return registres;
    }

    /**
     * Obtenir el nom i el tipus de cada camp d'una consulta.
     * @param rs Conjunt de registres retornats per la consulta
     * @return Array amb el nom i el tipus de cada camp: [][0] --> nom; [][1] --> tipus
     */
    public static String[][] getRsFields(ResultSet rs) {
        String[][] s = null;

        try {
            ResultSetMetaData rsmd = rs.getMetaData();
            int nCols = rsmd.getColumnCount();
            s = new String[nCols][2];
            for (int i = 0; i < nCols; i++) {
                s[i][0] =  rsmd.getColumnName(i + 1);
                s[i][1] =  rsmd.getColumnTypeName(i + 1);
            }
        } catch (SQLException throwables) {
            throwables.printStackTrace();
        }

        return s;
    }

    /**
     * Obtenir el màxim de caràcters que hi ha en cada camp.
     * @param data Conjunt de dades
     * @return Array d'enters amb l'amplada màxima en caràcters de cada camp
     */
    public static int[] getMaxChars(String[][] data, String[] fields, boolean includeFields) {
        int[] w = null;
        int c;
        int i;

        if (includeFields && fields != null) {
            c = fields.length;
            w = new int[c];
            for (i = 0; i < c; ++i) {
                w[i] = (fields[i]).length();
            }
        }

        if (data != null && data.length > 0) {
            c = data[0].length;
            if (w == null) w = new int[c];
            for (String[] d : data) {
                for (i = 0; i < c; ++i) {
                    if (d[i] != null && w[i] < (d[i]).length()) {
                        w[i] = (d[i]).length();
                    }
                }
            }
        }

        return w;
    }

}
