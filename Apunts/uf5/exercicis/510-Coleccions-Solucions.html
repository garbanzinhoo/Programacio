import java.util.ArrayList;
import java.util.HashMap;

public class Zoo {
    HashMap<String, Animal> animals = new HashMap<>();

    public HashMap getAnimals() {
        return animals;
    }

    public void afegirAnimal(Animal animal) {
        if (animal.getGrup() == null || animal.getAlimentacio() == null) return;

        Animal a = buscarAnimal(animal.getNom());

        if (a == null) {
            animals.put(animal.getNom(), animal);
        }
    }

    public Animal buscarAnimal(String nom) {
        return animals.get(nom);
    }

    public int quantsTipus() {
        return animals.size();
    }

    public ArrayList filtrarGrup(String grup) {
        ArrayList<Animal> fa = new ArrayList<>();
        Animal a;

        // Utilitzant keySet()
        for (String nom: animals.keySet()) {
            a = animals.get(nom);
            if (a.getGrup().equals(grup)) {
                fa.add(a);
            }
        }

        return fa;
    }

    public ArrayList filtrarAlimentacio(String alimentacio) {
        ArrayList<Animal> fa = new ArrayList<>();

        // Utilitzant values()
        for (Animal a: animals.values()) {
            if (a.getAlimentacio().equals(alimentacio)) {
                fa.add(a);
            }
        }

        return fa;
    }
}

final static String[] numeros = {"Zero", "Un", "Dos", "Tres", "Quatre", "Cinc", "Sis", "Set", "Vuit", "Nou", "Deu"};

public static void main(String[] args) {
    Stack<String> pila = new Stack<>();

    omplirPila(pila, 5);
    compteEnrere(pila);

    omplirPila(pila, 12);
    compteEnrere(pila);
}

// Posar el números en la pila (mínim 1, màxim 10)
static void omplirPila(Stack p, int n) {
    if (n < 1) n = 1;
    if (n > 10) n = 10;

    for (int i = 0; i <= n; i++) {
        p.push(numeros[i]);
    }
}

// Fer el compte enrere
static void compteEnrere(Stack p) {
    System.out.println("\n" + p.pop());
    try {
        while (!p.empty()) {
            Thread.sleep(1000);
            System.out.println(p.pop());
        }
    } catch (Exception e) {
        e.printStackTrace();
    }
}

        }

import java.time.LocalDateTime;

public class Compra implements Comparable {
    boolean prime;
    String nom;
    String adresa;
    LocalDateTime data;

    public Compra(boolean p, String n, String a , LocalDateTime d) {
        prime = p;
        nom = n;
        adresa = a;
        data = d;
    }

    @Override
    public String toString() {
        return (prime ? "\nClient \"Prime\"" : "") +
                "\nNom: " + nom +
                "\nAdreça: " + adresa +
                "\nnData: " + data;
    }

    @Override
    public int compareTo(Object o) {
        if (this.prime && !((Compra)o).prime) return -1;
        if (!this.prime && ((Compra)o).prime) return 1;
        return this.data.compareTo(((Compra)o).data);
    }
}

import java.time.LocalDateTime;
import java.util.PriorityQueue;
import java.util.Scanner;

public class E6CuaPrioritaria {
    static Scanner scan = new Scanner(System.in);

    public static void main(String[] args) {
        PriorityQueue<Compra> compres = new PriorityQueue<>();
        String opcio;

        opcio = menu();
        while (!opcio.equals("3")) {
            switch (opcio) {
                case "1":
                    compres.add(comprar());
                    break;
                case "2":
                    enviar1(compres);
                    break;
                default:
                    System.out.println("Opció incorrecta");
            }
            opcio = menu();
        }

        mostrarPendents(compres);
        System.out.println("\nAdeu!");
    }

    // Mostrar menú i demanar opció
    static String menu() {
        System.out.println("\n1 - Comprar");
        System.out.println("2 - Enviar");
        System.out.println("3 - Tancar");
        System.out.print("Tria una opció: ");

        return scan.nextLine();
    }

    // Demanar dades de la compra
    static Compra comprar() {
        String t;
        boolean p;
        String n;
        String a;

        System.out.print("És un client \"Prime\" (S/N)? ");
        t = scan.nextLine();
        p = t.equalsIgnoreCase("S");

        System.out.print("Nom: ");
        n = scan.nextLine();

        System.out.print("Adreça: ");
        a = scan.nextLine();

        return new Compra(p, n, a, LocalDateTime.now());
    }

    // Envia la primera compra de la cua
    static void enviar1(PriorityQueue c) {
        if (c.isEmpty()) {
            System.out.println("No queden compres per enviar");
        } else {
            System.out.println("S'ha enviat la compra:" + c.remove());
        }
    }

    // Mostrar (i treure) compres pendents d'enviar
    static void mostrarPendents(PriorityQueue c) {
        if (c.isEmpty()) {
            System.out.println("No queda cap comanda per enviar");
        }

        System.out.println("Pendents:");
        while (!c.isEmpty()) {
            System.out.println(c.remove());
        }
    }
}
