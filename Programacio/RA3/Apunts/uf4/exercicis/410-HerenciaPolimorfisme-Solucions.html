public class Data {
    private int dia, mes, any;

    public Data(int dia, int mes, int any) {
        set(dia, mes, any);
    }

    public Data(String data) {
        if (!data.matches("[0-9][0-9]-[0-9][0-9]-[0-9]{4}")) throw new Error("Eformat de la data ha de ser DD-MM-AAAA");
        String[] dn = data.split("-");
        set(Integer.parseInt(dn[0]), Integer.parseInt(dn[1]), Integer.parseInt(dn[2]));
    }

    private void set(int dia, int mes, int any) {
        if (!verificar(dia, mes, any)) throw new Error("El dia,mes o any no són correctes");
        this.dia = dia;
        this.mes = mes;
        this.any = any;
    }

    private boolean verificar(int dia, int mes, int any) {
        // TODO: verificar dies segons el mes i l'any
        return !(dia < 1 || dia > 31 || mes < 1 || mes > 31 || any < 1900 || any > 2023);
    }

    @Override
    public String toString() {
        String d = "";
        if (dia < 10) d += "0";
        d += dia + "-";
        if (mes < 10) d += "0";
        d += mes + "-";
        return d += any;
    }
}

Ordinador

public class Ordinador {
    // Atributs
    String nom;
    Component[] components;

    // Constructor
    public Ordinador(String nom) {
        this.nom = nom;
        components = new Component[10];
    }

    // Getters
    public float getPreu() {
        float p = 0;

        for (Component c: components)
            if (c != null) p += c.preu;

        return p;
    }

    public boolean afegirComponent(Component comp) {
        int ndx = posicioBuida();

        if (ndx < 0) return false;
        if (existeixComponent(comp)) return false;

        components[ndx] = comp;
        return true;
    }

    public boolean treureComponent(Component comp) {
        int ndx = posicioComponent(comp);

        if (ndx < 0) return false;

        components[ndx] = null;
        return true;
    }

    private int posicioBuida() {
        for (int i = 0; i < components.length; ++i) {
            if (components[i] == null) return i;
        }
        return -1;
    }

    private boolean existeixComponent(Component comp) {
        for (Component c: components) {
            if (c != null && c.getClass() == comp.getClass()) return true;
        }
        return false;
    }

    private int posicioComponent(Component comp) {
        for (int i = 0; i < components.length; ++i) {
            if (components[i] != null && components[i].getClass() == comp.getClass()) return i;
        }
        return -1;
    }

    private boolean ordinadorComplet() {
        int nc = 0;
        for (Component c: components) {
            if (c != null) ++nc;
        }
        return nc == 9;
    }

    @Override
    public String toString() {
        String s = "   ---" + nom + "---";

        for (Component c: components) {
            if (c != null) s += c;
        }

        if (!ordinadorComplet()) s += "\n--- L'ordinador no està complet!!! ---";
        else s += "\n--- Preu total: " + getPreu() + " ---";

        return s;
    }
}

PoligonRegular

public class PoligonRegular {
    String nom;
    double radi;
    int nCostats;

    double costat;
    double apotema;

    public PoligonRegular(String nom) {
        this.nom = nom;
    }

    public PoligonRegular(String nom, double radi, int nCostats) {
        this.nom = nom;
        this.radi = radi;
        this.nCostats = nCostats;
        calcular();
    }

    private void calcular() {
        double angle = 2 * Math.PI / nCostats;
        double base = radi - radi * Math.cos(angle);
        double altura = radi * Math.sin(angle);
        costat = Math.sqrt(base * base + altura * altura);
        apotema = Math.sqrt(radi * radi - costat * costat / 4);
    }

    private boolean similar(PoligonRegular pr) {
        double v1 = calcularArea();
        double v2 = pr.calcularArea();
        if (Math.abs(v1 - v2) < 0.1) return true;

        v1 = calcularPerimetre();
        v2 = pr.calcularPerimetre();
        if (Math.abs(v1 - v2) < 0.1) return true;

        return false;
    }

    public double calcularArea() {
        return (nCostats * costat * apotema) / 2;
    }

    public double calcularPerimetre() {
        return nCostats * costat;
    }

    @Override
    public boolean equals(Object o) {
        return nom.equals(((PoligonRegular)o).nom) || similar(((PoligonRegular)o));
    }

    @Override
    public String toString() {
        return nom +
                ", radi=" + radi +
                ": nCostats=" + nCostats +
                ": costat=" + costat +
                ": apotema=" + apotema +
                ", area=" + calcularArea() +
                ", perímetre=" + calcularPerimetre();
    }
}

TriangleEquilater

public class TriangleEquilater extends PoligonRegular {
    public TriangleEquilater(double base) {
        super("Triangle");
        this.costat = base;
        this.apotema = base * Math.sin(Math.PI / 3);
    }

    // Mètodes
    @Override
    public double calcularArea() { return (costat * apotema) / 2; }

    @Override
    public double calcularPerimetre() {
        return costat * 3;
    }

    @Override
    public String toString() {
        return nom +
                ": base=" + costat +
                ", altura=" + apotema +
                ", area=" + calcularArea() +
                ", perímetre=" + calcularPerimetre();
    }
}
