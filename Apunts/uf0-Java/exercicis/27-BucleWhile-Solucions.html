        if (resposta > mesGran) mesGran = resposta;

    System.out.println("No has introduït cap número");

int nota, suspensos = 0, total = 0, numNotes = 0;

do {
    System.out.print("Introdueix una nota entre 1 i 10 (0 per acabar): ");
    nota = scan.nextInt();

    if (nota > 0 && nota <= 10) {
        total += nota;
        ++numNotes;
        if (nota < 5) ++suspensos;
    } else {
        if (nota != 0) System.out.println("La nota és incorrecta");
    }
} while (nota != 0);

if (numNotes > 0) {
    System.out.printf("\nSuspensos: %d\n", suspensos);
    System.out.printf("Mitjana: %.1f\n", (float)total / numNotes);
} else {
    System.out.println("No has introduït cap nota");
}
