public abstract class Poligon {
    // Atributs
    String nom;

    // Constructor
    public Poligon(String nom) {
        this.nom = nom;
    }

    // Mètode estàtic per calcular la hipotenusa
    public static double calcularHipotenusa(double catet1, double catet2) {
        //return Math.sqrt(Math.pow(catet1, 2) + Math.pow(catet2, 2));
        return Math.hypot(catet1, catet2);
    }

    // Mètodes abstractes
    public abstract double calcularArea();

    public abstract double calcularPerimetre();

    // Getters calculats
    public double getArea() {
        return calcularArea();
    }

    public double getPerimetre() {
        return calcularPerimetre();
    }
}

Classe PoligonRegular

public class PoligonRegular extends Poligon {
    double radi;
    int nCostats;

    double costat;
    double apotema;

    public PoligonRegular(String nom, double radi, int nCostats) {
        super(nom);
        this.radi = radi;
        this.nCostats = nCostats;
        calcular();
    }

    private void calcular() {
        double angle = 2 * Math.PI / nCostats;
        double base = radi - radi * Math.cos(angle);
        double altura = radi * Math.sin(angle);
        costat = Math.sqrt(base * base + altura * altura);
        apotema = Math.sqrt(radi * radi - costat * costat / 4);
    }

    // Mètodes
    @Override
    public double calcularArea() {
        return (nCostats * costat * apotema) / 2;
    }

    @Override
    public double calcularPerimetre() {
        return nCostats * costat;
    }

    @Override
    public String toString() {
        return nom +
                ": nCostats=" + nCostats +
                ", lCostat=" + costat +
                ", apotema=" + apotema +
                ", area=" + getArea() +
                ", perímetre=" + getPerimetre();
    }
}

public abstract class Poligon implements Comparable<Poligon> {
    // Atributs
    String nom;

    // Constructor
    public Poligon(String nom) {
        this.nom = nom;
    }

    // Mètode estàtic per calcular la hipotenusa
    public static double calcularHipotenusa(double catet1, double catet2) {
        return Math.sqrt(Math.pow(catet1, 2) + Math.pow(catet2, 2));
    }

    // Mètodes abstractes
    public abstract double calcularArea();

    public abstract double calcularPerimetre();

    @Override
    public boolean equals(Object poligon) {
        return ((Poligon)poligon).getArea() == this.getArea();
    }

    public static boolean equals(Poligon p1, Poligon p2) {
        return p1.equals(p2);
    }

    // Comparable
    @Override
    public int compareTo(Poligon p) {
        return Double.compare(this.calcularArea(), p.calcularArea());
        //if (this.calcularArea() < p.calcularArea()) return -1;

        //else return 0;
    }

    public static int compareTo(Poligon p1, Poligon p2) {
        return p1.compareTo(p2);
    }

    // Getters calculats
    public double getArea() {
        return calcularArea();
    }

import java.util.Scanner;

public class Joc {
    static final int FILES = 20;

    static final int COLUMNES = 20;

    private static Scanner scan = new Scanner(System.in);

    private static boolean fi;

    private static Pocio pocio;

    private static Jugador asterix;

    private static Enemic[] legionaris = new Enemic[4];

    private static Taulell bosc;

    public static void main(String[] args) {
        bosc = new Taulell(FILES, COLUMNES);
        pocio = new Pocio('P', posicioAleatoria());
        asterix = new Jugador('J', posicioAleatoria());
        for (int i = 0; i < legionaris.length; ++i) {
            legionaris[i] = new Enemic((char)('1' + i), posicioAleatoria());
        }
        fi = false;

        do {
            bosc.init(pocio, asterix, legionaris);
            bosc.dibu();
            fi = mouJugador();
            if (!fi) fi = mouEnemics();
        } while(!fi);
    }

    public static boolean mouEnemics() {
        int dir;
        for (int i = 0; i < legionaris.length; ++i) {
            // TODO: s'haurien de moure cap l'Asterix
            dir = (int)(Math.random() * 4);
            switch(dir) {
                case 0:
                    legionaris[i].mouL();
                    break;
                case 1:
                    legionaris[i].mouR();
                    break;
                case 2:
                    legionaris[i].mouU();
                    break;
                case 3:
                    legionaris[i].mouD();
                    break;
            }
        }
        return false;
    }

    public static boolean mouJugador() {
        char op;

        do {
            System.out.println("A - Left");
            System.out.println("D - Right");
            System.out.println("W - Up");
            System.out.println("S - Down");
            System.out.println("Z - Exit");
            op = scan.nextLine().toUpperCase().charAt(0);
        } while(!(op == 'A' || op == 'D' || op == 'W' || op == 'S' || op == 'Z'));

        switch(op) {
            case 'A':
                asterix.mouL();
                break;
            case 'D':
                asterix.mouR();
                break;
            case 'W':
                asterix.mouU();
                break;
            case 'S':
                asterix.mouD();
                break;
        }

        return op == 'Z';
    }

    public static Coordenades posicioAleatoria() {
        return new Coordenades(1, FILES - 2, 1, COLUMNES - 2);
    }
}

        this.columna = columna;
    }

    public Coordenades(int minF, int maxF, int minC, int maxC) {
        fila = (int)(Math.random() * (maxF - minF + 1) + minF);

Taulell

public class Taulell {
    private int files;

    private int columnes;

    private static char[][] taulell;

    public Taulell(int files, int columnes) {
        this.files = files;
        this.columnes = columnes;
        taulell = new char[files][columnes];
    }

    public void init(Pocio p, Jugador jug, Enemic[] enemics) {
        for (int i = 0; i < files; ++i) {
            for (int j = 0; j < columnes; ++j) {
                if (i == 0 || j == 0 || i == files -1 || j == columnes - 1) taulell[i][j] = '#';
                else taulell[i][j] = ' ';
            }
        }
        taulell[p.coord.fila][p.coord.columna] = p.nom;
        taulell[jug.coord.fila][jug.coord.columna] = jug.nom;
        for (Enemic e: enemics) {
            taulell[e.coord.fila][e.coord.columna] = e.nom;
        }
    }

    public void dibu() {
        for (int i = 0; i < files; ++i) {
            for (int j = 0; j < columnes; ++j) {
                System.out.print(" " + taulell[i][j] + " ");
            }
            System.out.println();
        }
    }
}

Personatge (abstracta, implementa Moviment)

abstract class Personatge extends Element implements Moviment {
    int velocitat;

    int energia;

    public Personatge(char nom, Coordenades c, int velocitat) {
        super(nom, c);
        this.velocitat = velocitat;
    }

    public void mouL() {
        coord.columna -= velocitat;
        if (coord.columna < 1) coord.columna = 1;
        if (energia > 0) --energia;
    }

    public void mouR() {
        coord.columna += velocitat;
        if (coord.columna >= Joc.COLUMNES - 1) coord.columna = Joc.COLUMNES - 2;
        if (energia > 0) --energia;
    }

    public void mouU() {
        coord.fila -= velocitat;
        if (coord.fila < 1) coord.fila = 1;
        if (energia > 0) --energia;
    }

    public void mouD() {
        coord.fila += velocitat;
        if (coord.fila >= Joc.FILES - 1) coord.fila = Joc.FILES - 2;
        if (energia > 0) --energia;
    }

    public void setVelocitat(int velocitat) {
        this.velocitat = velocitat;
    }
}
