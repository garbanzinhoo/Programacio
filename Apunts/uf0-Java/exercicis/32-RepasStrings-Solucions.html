String s = "1234-ABC";
int ndx = s.indexOf("-");

if (s.length() != 8 || ndx < 0 || ndx != 4) {
    System.out.println("Error en la matrícula");
    return;
}

String s1 = s.substring(0, ndx);
for (int i = 0; i < s1.length(); ++i) {
    if (s1.charAt(i) < '0' || s1.charAt(i) > '9') {
        System.out.println("Error en la matrícula");
        return;
    }
}

s1 = s.substring(ndx + 1);
for (int i = 0; i < s1.length(); ++i) {
    if (s1.charAt(i) < 'A' || s1.charAt(i) > 'Z') {
        System.out.println("Error en la matrícula");
        return;
    }
}

System.out.println("Matrícula OK");
