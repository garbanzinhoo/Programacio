function mostrarArrayBi(array) {
    let resultat = "";
    for (let i = 0; i < array.length; i++) {
        for (let j = 0; j < array[i].length; j++) {
            resultat += ("" + array[i][j]).padStart(3, "\xa0");
        }
        resultat += "<br>";
    }
    terminal.innerHTML += resultat;
}

function multiplicarArraysBi(a1, a2) {
    // Obtenir les files i columnes de cada matriu
    let f1 = a1.length;
    let c1 = a1[0].length;
    let f2 = a2.length;
    let c2 = a2[0].length;

    // Comprovar que les columnes de la 1ª siguin igual a les files de la 2ª
    if (c1 != f2) return null;

    // Crear matriu resultat
    let i, j, k, t;
    let a3 = [];

    // Multiplicar matrius
    for (i = 0; i < f1; ++i) {
        a3[i] = [];
        for (j = 0; j < c2; ++j) {
            t = 0;
            for (k = 0; k < c1; ++k) {
                t += a1[i][k] * a2[k][j];
            }
            a3[i][j] = t;
        }
    }

    return a3;
}

"use strict";

const iMida = document.getElementById("mida");
const bInicialitzar = document.getElementById("inicialitzar");
const bSeguent = document.getElementById("seguent");
const terminal = document.getElementById("terminal");

bInicialitzar.addEventListener("click", init);
bSeguent.addEventListener("click", seguentGeneracio);

let generacio;
let actual;
let seguent;
let mida;

function init() {
    mida = parseInt(iMida.value);
    generacio = 0;
    actual = [];
    seguent = [];

    // Inicialitzar a 0
    for (let i = 0; i < mida; ++i) {
        actual[i] = [];
        seguent[i] = [];
        for (let j = 0; j < mida; ++j) {
            actual[i][j] = 0;
            seguent[i][j] = 0;
        }        
    }

    // Posar 1 aleatòriament
    for (let i = 1; i < mida - 1; ++i) {
        for (let j = 1; j < mida - 1; ++j) {
            if (Math.random() < .5) actual[i][j] = 1;
        }        
    }

    dibuixar(actual);
}

function seguentGeneracio() {
    for (let f = 1; f < mida - 1; f++) {
        for (let c = 1; c < mida - 1; c++) {
            // Comptar quants veïns
            let veins  = actual[f-1][c-1] + actual[f-1][c] + actual[f-1][c+1];
            veins += actual[f  ][c-1] +                  actual[f  ][c+1];
            veins += actual[f+1][c-1] + actual[f+1][c] + actual[f+1][c+1];

            // Implementar regles
            if ((actual[f][c] == 1) && (veins < 2 || veins > 3))
                seguent[f][c] = 0;    // Hi ha pocs veïns o massa: desapareix
            else if ((actual[f][c] == 0) && (veins == 3))
                seguent[f][c] = 1;    // Nou bacteri
            else
                seguent[f][c] = actual[f][c]; // No canvia
        }
    }

    ++generacio;

    copiar(actual, seguent);

    dibuixar(actual);
}

function dibuixar(actual) {
    let resultat = "Generació " + generacio + "<br><br>";
    for (let f = 0; f < mida; f++) {
        for (let c = 0; c < mida; c++) {
            resultat += actual[f][c] == 0 ? "·  " : "B  ";
        }
        resultat += "<br>";
    }
    terminal.innerHTML = resultat;
}

function copiar(actual, seguent) {
    for (let f = 1; f < mida - 1; f++)
        for (let c = 1; c < mida - 1; c++)
            actual[f][c] = seguent[f][c];
}

        char l;

        do {
            l = introLletra();
            PenjatModel.posarLletra(l);
        } while (!PenjatModel.haAcabat());

    static char introLletra() {
        char l = ' ';

        while (l < 'A' || l > 'Z') {

            l = scan.nextLine().toUpperCase().charAt(0);
        }

public class PenjatModel {
    static String[] paraules = {
            "MURCIELAGO",
            "ORDENADOR",
            "METROPOLITANO",
            "PROGRAMADOR",
            "QUITANIEVES",
            "VULCANOLOGO",
            "RASCACIELOS",
            "PENSAMIENTO",
            "DESCONOCIDO",
            "FANTASTICO"
    };

    static String paraula;
    static StringBuilder paraulaJugador;
    static int intents;
    static boolean fi;

    public static void triarParaula() {
        intents = 0;
        fi = false;

        paraula = paraules[(int)(Math.random() * paraules.length)];
        paraulaJugador = new StringBuilder("");
        for (int i = 0; i < paraula.length(); i++) paraulaJugador.append("_");

        PenjatVista.mostrarParaula(PenjatModel.paraulaJugador);
    }

    public static void posarLletra(char l) {
        if (paraula.contains("" + l)) {
            for (int i = 0; i < paraula.length(); i++) {
                if (paraula.charAt(i) == l) {
                    paraulaJugador.setCharAt(i, l);
                }
            }
            PenjatVista.mostrarParaula(PenjatModel.paraulaJugador);
            if (haGuanyat()) PenjatVista.mostrarMissatge("HAS GUANYAT!");
        } else {
            ++intents;
            PenjatVista.mostrarPenjat(intents);
            if (intents == 7) {
                fi = true;
                PenjatVista.mostrarMissatge("HAS PERDUT!");
                PenjatVista.mostrarMissatge("La paraula era " + paraula);
            } else {
                PenjatVista.mostrarParaula(PenjatModel.paraulaJugador);
            }
        }
    }

    public static boolean haGuanyat() {
        for (int i = 0; i < paraula.length(); i++) {
            if (paraulaJugador.charAt(i) == '_') return false;
        }
        fi = true;

        return true;
    }

    public static boolean haAcabat() {
        return fi;
    }
}

Model

public class SetMigModel {

    static final String[] pals = {"Ors", "Cors", "Espases", "Bastos"};
    static final String[] figures = {"1", "2", "3", "4", "5", "6", "7",
                                     "Sota", "Cavall", "Rei"};
    static String[] cartes = new String[pals.length * figures.length];

    static float puntsJugador;
    static float puntsBanca;
    static int carta;
    static boolean fi;

    public static void inicialitzar() {
        int k = 0, r;
        String s;

        // Generar baralla
        for (int i = 0; i < pals.length; i++) {
            for (int j = 0; j < figures.length; j++) {
                cartes[k] = figures[j] + " de " + pals[i];
                ++k;
            }
        }

        // Barrejar cartes
        for (k = 0; k < cartes.length; k++) {
            r = (int)(Math.random() * cartes.length);
            s = cartes[k];
            cartes[k] = cartes[r];
            cartes[r] = s;
        }

        // Inicialitzar valors
        puntsJugador = 0;
        puntsBanca = 0;
        carta = 0;
        fi = false;
    }

    public static void treurePlantar(char l) {
        if (l == 'T') {
            puntsJugador += treuCarta();
            SetMigVista.mostrarJugadorPunts("El jugador", puntsJugador);
            if (puntsJugador > 7.5) {
                SetMigVista.mostrarMissatge("HAS PERDUT!");
                fi = true;
            }
        } else {
            banca();
        }
    }

    public static void banca() {
        while (puntsJugador > puntsBanca) {
            puntsBanca += treuCarta();
            SetMigVista.mostrarJugadorPunts("La banca", puntsBanca);
            try {
                Thread.sleep(3000);
            } catch (InterruptedException e) {
            }
        }

        if (puntsJugador == puntsBanca)
            SetMigVista.mostrarMissatge("EMPAT: guanya la banca");
        else {
            if (puntsBanca > 7.5) SetMigVista.mostrarMissatge("HAS GUANYAT!");
            else SetMigVista.mostrarMissatge("HAS PERDUT!");
        }
        fi = true;
    }

    public static float treuCarta() {
        String c = cartes[carta];
        float punts;

        if (c.charAt(0) >= '1' && c.charAt(0) <= '7') {
            punts = c.charAt(0) - '0';
        } else {
            punts = 0.5f;
        }
        SetMigVista.mostrarCartaPunts(c, punts);
        ++carta;

        return punts;
    }

    public static boolean haAcabat() {
        return fi;
    }

}
