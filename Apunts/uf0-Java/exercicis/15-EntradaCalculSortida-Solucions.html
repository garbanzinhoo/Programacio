float base, altura;
float area, hipotenusa, perimetre;

// Introduir dades
System.out.print("Base del triangle rectangle: ");
base = scan.nextFloat();
scan.nextLine();
System.out.print("Altura del triangle rectangle: ");
altura = scan.nextFloat();
scan.nextLine();

// Calcular àrea, hipotenusa i perímetre
area = base * altura / 2;
hipotenusa = (float)Math.sqrt((base * base) + (altura * altura));
perimetre = base + altura + hipotenusa;

// Mostrar resultats
System.out.printf("Àrea: %.3f\nPerímetre: %.3f\n", area, perimetre);

//            per trobar els minutsT i segons
minutsTotals = segonsTotals / 60;
segons =       segonsTotals % 60;
//  i divisió i residu de minutsT / 60 per trobar hores i minuts
hores =  minutsTotals / 60;
minuts = minutsTotals % 60;
 
// Solució 2: basada en calcular minutsT i hores
//            dividint per segonsT entre 60 i 3600
minutsTotals = segonsTotals / 60;
hores = segonsTotals / 3600;    // També es pot fer hores = minutsTotals / 60;
//  i trobar els minuts i segons restant hores i minuts convertits a segons
minuts = minutsTotals - (hores * 60);
segons = segonsTotals - (minuts * 60) - (hores * 3600);
