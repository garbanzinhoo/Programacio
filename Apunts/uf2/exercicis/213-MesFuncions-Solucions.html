// CONVERTIR SEGONS
static void convertirSegons() {
    System.out.print("Introdueix els segons: ");
    int segons = introIntMinMax(0, 360000 - 1);
    String hms = formatarHMS(segons);
    mostrarHMS(segons, hms);
}

// Introduir un número enter
public static int introInt() {
    int valor;

    while (!scan.hasNextInt()) {
        scan.nextLine();
        System.out.println("Error: has d'introduir un número enter");
    }
    valor = scan.nextInt();
    scan.nextLine();

    return valor;
}

// Introduir un número enter entre dos valors
public static int introIntMinMax(int min, int max) {
    int valor;

    if (min > max) {
        int t = min;
        min = max;
        max = t;
    }

    valor = introInt();
    while (valor < min || valor > max) {
        System.out.println("Error: has d'introduir un número entre " + min + " i " + max);
        valor = introInt();
    }

    return valor;
}

static String formatarHMS(int segonsTotals) {
    int minutsTotals;
    int segons, minuts, hores;

    minutsTotals = segonsTotals / 60;
    segons =       segonsTotals % 60;
    hores =  minutsTotals / 60;
    minuts = minutsTotals % 60;

    return String.format("%02d:%02d:%02d", hores, minuts, segons);
}

static void mostrarHMS(int segons, String hms) {
    System.out.println(segons + " segons equivalen a " + hms);
}

// CALCULAR AUTONOMIA
static void calcularAutonomia() {
    float capacitat = introReal("Introdueix la capacitat: ");
    float consum = introReal("Introdueix el consum cada 100 Km: ");
    int[] autonomia = calcularAutonomia(capacitat, consum);
    mostrarAutonomia(autonomia);
}

// Introduir un número real
public static float introReal(String pregunta) {
    float valor;

    System.out.print(pregunta);
    while (!scan.hasNextFloat()) {
        scan.nextLine();
        System.out.println("Error: has d'introduir un número real");
    }
    valor = scan.nextFloat();
    scan.nextLine();

    return valor;
}

static int[] calcularAutonomia(float capacitat, float consum100) {
    float consumM, consumKm;
    int metresTotals, metres, km;
    int[] auto = new int[2];

    // Calcular les unitats més petites (consum / m)
    consumKm = consum100 / 100;     // Consum per Km
    consumM = consumKm / 1000;      // Consum per metre

    metresTotals = (int)(capacitat / consumM);
    km =     metresTotals / 1000;
    metres = metresTotals % 1000;

    auto[0] = km;
    auto[1] = metres;
    return auto;
}

static void mostrarAutonomia(int[] auto) {
    System.out.println("El vehicle podrà fer " + auto[0] + " Km i " + auto[1] + " m");
}

// ÉS ANY DE TRASPÀS
static void anyTraspas() {
    mostrarAnyTraspas(2000, esAnyTraspas(2000));
    mostrarAnyTraspas(2020, esAnyTraspas(2020));
    mostrarAnyTraspas(2100, esAnyTraspas(2100));
    mostrarAnyTraspas(2222, esAnyTraspas(2222));
}

static boolean esAnyTraspas(int any) {
    if (any % 400 == 0) return true;
    else if (any % 100 == 0) return false;
    else if (any % 4 == 0) return true;
    else return false;
}

static void mostrarAnyTraspas(int any, boolean esTraspas) {
    System.out.println("L'any " + any + (esTraspas ? "" : " NO") + " és any de traspàs");
}

// DIA SETMANA
static void diaSetmana() {
    System.out.println(diaSetmana(1));
    System.out.println(diaSetmana(3));
    System.out.println(diaSetmana(6));
    System.out.println(diaSetmana(8));
}

static String diaSetmana(int dia) {
    if (dia >= 1 && dia <= 7) {
        if (dia == 1) return "Dilluns";
        else if (dia == 2) return "Dimarts";
        else if (dia == 3) return "Dimecres";
        else if (dia == 4) return "Dijous";
        else if (dia == 5) return "Divendres";
        else if (dia == 6) return "Dissabte";
        else return "Diumenge";
    } else {
        return "El dia ha d’estar entre 1 i 7";
    }
}

// DIES MES
static void diesMes() {
    mostrarDiesMes(diesMes(1, 2000), 1, 2000);
    mostrarDiesMes(diesMes(2, 2020), 2, 2020);
    mostrarDiesMes(diesMes(4, 2100), 4, 2100);
    mostrarDiesMes(diesMes(2, 2222), 2, 2222);
}

static int diesMes(int mes, int any) {
    if (mes >= 1 && mes <= 12) {
        if (mes == 1 || mes == 3 || mes == 5 || mes == 7 || mes == 8 || mes == 10 || mes == 12)
            return 31;
        else if (mes == 4 || mes == 6 || mes == 9 || mes == 11)
            return 30;
        else {
            if (any % 400 == 0 || (any % 4 == 0 && any % 100 != 0))
                return 29;
            else
                return 28;
        }
    } else {
        return -1;
    }
}

static void mostrarDiesMes(int dies, int mes, int any) {
    if (dies == -1) System.out.println("El mes ha d'estar entre 1 i 12");
    else System.out.println("El mes " + mes + " de l'any " + any + " te " + dies + " dies");
}
