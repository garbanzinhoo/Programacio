"use strict";

const articles = [
    {
        codi: "123",
        descripcio: "Fregidora d'aire",
        preu: 123
    },
    {
        codi: "456",
        descripcio: "Pantalla OLED 50\"",
        preu: 456
    },
    {
        codi: "789",
        descripcio: "PC Gaming",
        preu: 789
    }
];

let cistella = [];

const iCodi = document.getElementById("codi");
const bAFegir = document.getElementById("afegir");
const bTreure = document.getElementById("treure");
const bPagar = document.getElementById("pagar");
const terminal = document.getElementById("terminal");

bAFegir.addEventListener("click", afegirArticle);
bTreure.addEventListener("click", treureArticle);
bPagar.addEventListener("click", pagarCompra);

function afegirArticle() {
    // Agafar el codi
    const codi = iCodi.value.trim();

    // Buscar el codi en l'array d'articles
    let ndx = -1;
    for (let i = 0; i < articles.length; i++) {
        if (articles[i].codi === codi) {
            ndx = i;
            break;
        }
    }

    // Si no l'ha trobat, error i sortir
    if (ndx === -1) {
        alert("Codi inexistent");
        return;
    }

    // Afegir l'article a la cistella
    cistella.push(articles[ndx]);
}

function treureArticle() {
    // Agafar el codi
    const codi = iCodi.value.trim();

    // Buscar el codi en la cistella
    let ndx = -1;
    for (let i = 0; i < cistella.length; i++) {
        if (cistella[i].codi === codi) {
            ndx = i;
            break;
        }
    }

    // Si no l'ha trobat, error i sortir
    if (ndx === -1) {
        alert("Codi inexistent");
        return;
    }

    // Treure l'article de la cistella
    cistella.splice(ndx, 1);
}

function pagarCompra() {
    let html = "";
    let total = 0;
    for (const a of cistella) {
        html += `${a.descripcio.padEnd(30, ".")} ${a.preu.toFixed(2).replace(".", ",").padStart(10)} €<br>`;
        total += a.preu;
    }

    html += `TOTAL: ${total.toFixed(2).replace(".", ",")} €`;

    // Mostrar resultat
    terminal.innerHTML = html;

    // Buidar cistella
    cistella = [];
    //cistella.length = 0;
}

"use strict";

const llibres = [];

const iISBN = document.getElementById("isbn");
const iTitol = document.getElementById("titol");
const iAutor = document.getElementById("autor");
const iGenere = document.getElementById("genere");
const iResum = document.getElementById("resum");

const bAFegir = document.getElementById("afegir");
const bLlista = document.getElementById("llista");
const bTematica = document.getElementById("tematica");

const terminal = document.getElementById("terminal");

bAFegir.addEventListener("click", afegirLlibre);
bLlista.addEventListener("click", llistaLlibres);
bTematica.addEventListener("click", llistaLlibresTematica);

function afegirLlibre() {
    // Agafar el codi
    const codi = iISBN.value.trim();

    // Buscar el codi en l'array de llibres
    let ndx = -1;
    for (let i = 0; i < llibres.length; i++) {
        if (llibres[i].isbn === codi) {
            ndx = i;
            break;
        }
    }

    // Si no l'ha trobat, error i sortir
    if (ndx !== -1) {
        alert("Aquest llibre ja existeix");
        return;
    }

    // Crear un objecte amb les dades del llibre i afegir-lo a la llista
    let l = {
        isbn: codi,
        titol: iTitol.value.trim(),
        autor: iAutor.value.trim(),
        genere: iGenere.value,
        resum: iResum.value.replaceAll("\n", "<br>")
    };

    llibres.push(l);
}

function llistaLlibres() {
    let html = "";
    for (const l of llibres) {
        html += `ISBN: ${l.isbn}<br>
        Títol: ${l.titol}<br>
        Autor: ${l.autor}<br>
        Gènere: ${l.genere}<br>
        Resum: ${l.resum}<br><br>`;
    }

    // Mostrar resultat
    terminal.innerHTML = html;
}

function llistaLlibresTematica() {
    const tema = iGenere.value.trim();
    let html = "";
    for (const l of llibres) {
        if (l.genere === tema) {
            html += `ISBN: ${l.isbn}<br>
            Títol: ${l.titol}<br>
            Autor: ${l.autor}<br>
            Gènere: ${l.genere}<br>
            Resum: ${l.resum}<br><br>`;
        }
    }

    // Mostrar resultat
    terminal.innerHTML = html;
}
