scan.nextLine();

max = notes[0];

    if (notes[i] < min) min = notes[i];
    if (notes[i] > max) max = notes[i];
}

    // Si alguna UF està suspesa, no aprova el mòdul

//  si un número és més petit que l'anterior, no ho és

    if (n[i] < n[i - 1]) {
        ordreAscendent = false;
        break;
    }
}

// Comprovar si ordre descendent:
//  si un número és més gran que l'anterior, no ho és
for (int i = 1; i < n.length; ++i) {
    if (n[i] > n[i - 1]) {
        ordreDescendent = false;
        break;
    }
}

// Si tots els números són iguals, ascendent i descendent!!!
if (ordreAscendent && ordreDescendent) {
    System.out.println("\nÉs constant!");
} else if (ordreAscendent) {
    System.out.println("\nOrdre ascendent");
} else if (ordreDescendent) {
    System.out.println("\nOrdre descendent");
} else {
    System.out.println("\nNo està ordenada");

char[] lletres = new char[10];
int l = lletres.length - 1;
char c;

// Demanar desplaçament
System.out.print("Desplaçament (1-9): ");
int desp = scan.nextInt();

// Generar i mostrar lletres aleatòries entre 'A' i 'Z'
for (int i = 0; i <= l; ++i) {
    lletres[i] = (char)(Math.random() * ('Z' - 'A' + 1) + 'A');
    System.out.print(lletres[i] + " ");
}
System.out.println();

// Rotar n vegades
while (desp > 0) {
    // Guardar l’última lletra
    c = lletres[l];
    // Desplaçar començant pel final
    for (int i = l; i > 0; i--) {
        lletres[i] = lletres[i - 1];
    }
    // Posar a la primera posició la que estava al final
    lletres[0] = c;

    --desp;
}

// Mostrar array rotat
for (int i = 0; i <= l; ++i) {
    System.out.print(lletres[i] + " ");
}
System.out.println();

System.out.println();

final int N = 20;
int[] numeros = new int[N];
int[] comptador = new int[10];
int c;

// Omplir array amb números aleatoris entre 0 i 9
for (int i = 0; i < N; ++i)
    numeros[i] = (int) (Math.random() * 10);

// Mostrar array
for (int i = 0; i < N; ++i)
    System.out.print(numeros[i] + " ");
System.out.println();

// SOLUCIÓ 1: array auxiliar
// L'array comptador[] té cada posició inicialitzada a 0
// Llegeix cada número i incrementa la posició corresponent de l'array comptador[]
for (int i = 0; i < N; ++i)
    ++comptador[numeros[i]];

// Per cada posició de l'array comptador[], mostra la posició (el número)
// i el seu contingut (les vegades que apareix)
for (int i = 0; i < comptador.length; ++i)
    System.out.println(i + " --> " + comptador[i] + " vegades");

// SOLUCIÓ 2: bucle niat
// Per cada valor entre 0 i 9...
for (int i = 0; i <= 9; ++i) {
    // ...inicialitzar el comptador
    c = 0;
    // Per cada número, si és igual que el valor de 'i', incrementar el comptador
    for (int j = 0; j < N; ++j)
        if (numeros[j] == i) ++c;
    // Mostrar el valor i les vegades que apareix
    System.out.println(i + " --> " + c + " vegades");
}
