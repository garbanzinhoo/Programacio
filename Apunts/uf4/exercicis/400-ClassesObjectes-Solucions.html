import java.time.LocalDate;

public class Persona {
    // Atributs privats
    private String nif;
    private String nom;
    private String dataNaixement;

    // Constructor
    public Persona(String nif, String nom, String naixement) {
        this.nom = nom;
        this.nif = nif;
        dataNaixement = naixement;
    }

    // Calcular l'edat a partir de la data de naixement i la data actual
    public int getEdat() {
        LocalDate avui = LocalDate.now();
        int any = avui.getYear();
        int mes = avui.getMonthValue();
        int dia = avui.getDayOfMonth();

        String[] dadesNaixement = dataNaixement.split("-");
        int diaN = Integer.parseInt(dadesNaixement[0]);
        int mesN = Integer.parseInt(dadesNaixement[1]);
        int anyN = Integer.parseInt(dadesNaixement[2]);

        int anys = any - anyN;
        if (mes < mesN || (mes == mesN && dia < diaN)) --anys;
        return anys;
    }

    @Override
    public String toString() {
        return nif + " : " + nom + " : " + getEdat();
    }
}

import java.time.LocalDate;
import java.time.Period;
import java.time.format.DateTimeFormatter;

public class PersonaV2 {
    // Atribut estàtic per comptar quantes persones s'han creat
    private static int comptadorPersones = 0;

    // Atributs privats
    private String nif;
    private String nom;
    private String dataNaixement;

    // Constructor
    public PersonaV2(String nif, String nom, String naixement) {
        // Verificar NIF i data de naixement
        // Si algun no és correcte, llança un error
        if (!verificarNif(nif)) throw new Error("NIF incorrecte");
        if (!verificarData(naixement)) throw new Error("Data de naixement incorrecte");

        // inicialitzar valors ja verificats
        // S'utilitza 'this' quan el nom de l'atribut coincideix amb el nom del paràmetre
        // (també es podria haver canviat el nom del paràmetre)
        this.nom = nom;
        this.nif = nif;
        dataNaixement = naixement;

        // Incrementem el comptador de persones creades
        ++comptadorPersones;
    }

    // Comprovar la data (caldria tenir en compte febrer, any de traspàs...)
    private boolean verificarData(String data) {
        String[] d = data.split("-");
        int dia = Integer.parseInt(d[0]);
        int mes = Integer.parseInt(d[1]);
        int any = Integer.parseInt(d[2]);

        return !(dia < 1 || dia > 31 || mes < 0 || mes > 12 || any < 1900 || any > 2020);
    }

    // Comprovar el NIF
    private boolean verificarNif(String nif) {
        if (nif.length() != 9) return false;

        char lletra = nif.charAt(8);
        if (lletra < 'A' || lletra > 'Z') return false;

        if (lletra != calcularLletraNif(nif.substring(0, 8))) return false;

        return true;
    }

    // Calcular la lletra del NIF
    private char calcularLletraNif(String dni) {
        final String LLETRES = "TRWAGMYFPDXBNJZSQVHLCKE";

        int residu;
        char lletra;

        // Impedeix que es generi una excepció que aturi el programa
        // Si no es pot convertir el dni a número, llança un error
        try {
            residu = Integer.parseInt(dni) % 23;
        } catch (Exception e) {
            throw new Error("Error en el número del DNI");
        }

        lletra = LLETRES.charAt(residu);

        return lletra;
    }

    // Obtenir el número de persones creades
    public static int getNumPersones() {
        return comptadorPersones;
    }

    // Inverteix l'ordre de DD-MM-AAAA a AAAA-MM-DD o alrevès
    public static String invertirData(String data) {
        String[] d = data.split("-");
        return d[2] + "-" + d[1] + "-" + d[0];
    }

    // Calcular l'edat a partir de la data de naixement i una data qualsevol
    public static int getEdat(String dn) {
        LocalDate avui = LocalDate.now();
        //LocalDate naix = LocalDate.parse(dn, DateTimeFormatter.ofPattern("dd-MM-yyyy"));
        LocalDate naix = LocalDate.parse(invertirData(dn));

        Period p = Period.between(naix, avui);

        return p.getYears();
    }

    // Calcular l'edat a partir de la data de naixement i la data actual
    public int getEdat() {
        return getEdat(dataNaixement);
    }

    @Override
    public String toString() {
        return nif + " : " + nom + " : " + getEdat();
    }
}

Classe Article

public class Article {
    // Atributs dels objectes
    private String codi;
    private String descripcio;
    private int unitats;
    private float preu;
    private int iva;

    // Constructor
    public Article(String _codi, String _descripcio, int _unitats, float _preu, int _iva) {
        // Verificar dades
        if (_codi.length() != 4) throw new Error("El codi ha e tenir 4 caràcters");
        if (_unitats < 0) throw new Error("Les unitats no poden ser negatives");
        if (_preu < .1) throw new Error("El preu no pot ser inferior a 10 cèntims");
        if (_iva != 0 && _iva != 4 && _iva != 5 && _iva != 10 && _iva != 21) throw new Error("El tant per cent d'IVA només pot ser 0, 4, 5, 10 o 21");

        // Inicialitzar atributs
        codi = _codi;
        descripcio = _descripcio;
        unitats = _unitats;
        preu = _preu;
        iva = _iva;
    }

    // Getters
    public String getCodi() { return codi; }
    public String getDescripcio() { return descripcio; }
    public int getUnitats() { return unitats; }
    public float getPreu() { return preu; }
    public int getIva() { return iva; }

    // Setters
    public boolean setUnitats(int _unitats) {
        if (_unitats < 0) return false;
        unitats = _unitats;
        return true;
    }

    public boolean setPreu(float _preu) {
        if (_preu < .1) return false;
        preu = _preu;
        return true;
    }

    public boolean setIva(int _iva) {
        if (_iva != 0 && _iva != 4 && _iva != 5 && _iva != 10 && _iva != 21) return false;
        iva = _iva;
        return true;
    }

    // Calcula i retorna el preu final
    public float getPreuFinal() {
        float subtotal = unitats * preu;
        return subtotal + subtotal * iva / 100;
    }

    @Override
    public String toString() {
        return codi + " : " + descripcio + " : " + String.format("%.2f €", getPreuFinal());
    }
}

import java.util.Scanner;

public class Cistella {

    // Articles de la cistella
    static Article[] productes = new Article[5];

    // Lector de dades (llegeix les dades introduïdes amb el teclat)
    static Scanner scan = new Scanner(System.in);

    // Mètode principal encarregat d'engegar l'aplicació
    public static void main(String[] args) {
        // Introduir articles
        for (int i = 0; i < productes.length; ++i) {
            productes[i] = introProducte();
        }

        // Modificar algunes dades (setters)
        productes[0].setUnitats(50);
        productes[1].setPreu(800);
        productes[4].setIva(10);

        // Mostrar els articles de la cistella i calcular el total
        float total = 0;
        for (Article p : productes) {
            System.out.println(p);
            total += p.getPreuFinal();
        }
        System.out.printf("Total de la cistella: %.2f €", total);
    }

    // Mètode per introduir un article
    private static Article introProducte() {
        Article p = null;

        String codi, descripcio;
        int unitats, iva;
        float preu;

        do {
            // Evita errors imprevistos amb la introducció de dades
            try {
                System.out.print("Introdueix el codi i la descripció: ");
                codi = scan.next().trim();
                descripcio = scan.nextLine().trim();
                System.out.print("Introdueix les unitats, el preu i el %iva: ");
                unitats = scan.nextInt();
                preu = scan.nextFloat();
                iva = scan.nextInt();
                p = creador(codi, descripcio, unitats, preu, iva);
                if (p != null) System.out.println(p);
            } catch (Exception e) {
                System.out.println(e.getMessage());
            }
        } while (p == null && scan.hasNext());

        return p;
    }

    // Mètode per crear un article tenint en compte possibles errors
    // Retorna un objecte de la classe Article o null si no s'ha pogut crear
    private static Article creador(String codi, String descripcio, int unitats, float preu, int iva) {
        Article p = null;

        // Evita errors amb les dades passades al constructor
        try {
            p = new Article(codi, descripcio, unitats, preu, iva);
        } catch (Error e) {
            System.out.println(e.getMessage());
        }

        return p;
    }
}

public class Ordinador {
    // Atributs
    String nom;
    Monitor monitor;
    Teclat teclat;
    Ratoli ratoli;
    Torre torre;

    // Constructor
    public Ordinador(String nom, Monitor monitor, Teclat teclat, Ratoli ratoli, Torre torre) {
        this.nom = nom;
        this.monitor = monitor;
        this.teclat = teclat;
        this.ratoli = ratoli;
        this.torre = torre;
    }

    // Getters
    public float getPreu() {
        return monitor.getPreu() + teclat.getPreu() + ratoli.getPreu() + torre.getPreu();
    }

    @Override
    public String toString() {
        return "   ---" + nom + "---" +
                "\n" + monitor.toString() +
                "\n" + teclat.toString() +
                "\n" + ratoli.toString() +
                "\n" + torre.toString();
    }
}
