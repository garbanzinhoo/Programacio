    try {

        if (!f.exists()) return;

        System.out.println("\n   Llegir totes les paraules");
        llegirParaules1(f);
        System.out.println("\n   Llegir fins que trobi ---");
        llegirParaules2(f);
        System.out.println("\n   Llegir N paraules i mostrar les de +5 lletres");

    } catch (Exception e) {
        e.printStackTrace();
    }
}

static void llegirParaules1(File f) throws FileNotFoundException {
    Scanner reader = new Scanner(f);

    while (reader.hasNext()) {
        System.out.println(reader.next());
    }

    reader.close();
}

static void llegirNumeros(String p) {
    try {
        File f = new File(p);
        if (!f.exists()) return;

        System.out.println("\n   Llegir tots els numeros");
        llegirNumeros1(f);
        System.out.println("\n   Llegir fins que trobi un negatiu");
        llegirNumeros2(f);
        System.out.println("\n   Llegir N numeros i ordenar");
        llegirNumeros3(f);

    } catch (Exception e) {
        e.printStackTrace();

}

static void llegirNumeros1(File f) throws FileNotFoundException {
    Scanner reader = new Scanner(f);
    int n = 0;

    while (reader.hasNext()) {
        System.out.print(reader.nextInt() + " ");
        ++n;
        if (n >= 6) {
            System.out.println();
            n = 0;
        }
    }
    if (n > 0) System.out.println();

    reader.close();
}

static void llegirNumeros2(File f) throws FileNotFoundException {
    Scanner reader = new Scanner(f);
    int v, n = 0;

    while (reader.hasNext()) {
        v = reader.nextInt();
        if (v < 0) break;

        ++n;
        System.out.print(v + " ");
        if (n >= 6) {
            System.out.println();
            n = 0;
        }
    }
    if (n > 0) System.out.println();

    reader.close();
}

static void llegirNumeros3(File f) throws FileNotFoundException {
    Scanner reader = new Scanner(f);
    int i = 0, j;
    int n = reader.nextInt();
    int[] numeros = new int[n];

    while (reader.hasNext() && i < n) {
        numeros[i] = reader.nextInt();
        ++i;
    }

    ordenar(numeros);

    i = 0; j = 0;
    while (i < n) {
        System.out.print(numeros[i] + " ");
        ++i; ++j;
        if (j >= 6) {
            System.out.println();
            j = 0;
        }
    }
    if (j > 0) System.out.println();

    reader.close();
}

static void ordenar(int[] n) {
    int t;

    for (int i = 0; i < n.length - 1; i++) {
        for (int j = i + 1; j < n.length; j++) {
            if (n[i] > n[j]) {
                t = n[i];
                n[i] = n[j];
                n[j] = t;
            }
        }
    }
}

static void llegirLinies(String p) {
    try {
        File f = new File(p);
        Scanner reader = new Scanner(f);

        System.out.println("\n   Llegir linies");
        while (reader.hasNext()) {
            if (llegirCamps(reader) && reader.hasNext()) System.out.println("------");
        }

        reader.close();
    } catch (Exception e) {
        e.printStackTrace();
    }
}

static boolean llegirCamps(Scanner reader) {
    String[] camps;
    String[] data;
    String linia;
    int any;

    linia = reader.nextLine().trim();
    camps = linia.split(";");
    if (camps.length != 3) return false;

    System.out.println(camps[0]);
    System.out.println("Data de naixement: " + camps[1]);
    any = Calendar.getInstance().get(Calendar.YEAR);
    data = camps[1].split("/");
    System.out.println("Edat aproximada: " + (any - Integer.parseInt(data[2])));
    System.out.println("Altura: " + camps[2]);
    return true;
}

static String getType(String s) {
    try {
        Double.parseDouble(s);
        try {
            long v = Long.parseLong(s);
            if (v >= Byte.MIN_VALUE && v <= Byte.MAX_VALUE)
                return "  Byte: " + v;
            if (v >= Short.MIN_VALUE && v <= Short.MAX_VALUE)
                return "    Short: " + v;
            if (v >= Integer.MIN_VALUE && v <= Integer.MAX_VALUE)
                return "      Integer: " + v;
            return "        Long: " + v;
        } catch (NumberFormatException e) {
            String d = "" + Double.parseDouble(s);
            String f = "" + Float.parseFloat(s);
            if (d.length() > f.length()) return "            Double: " + d;
            else return "          Float: " + f;
        }
    } catch (NumberFormatException e) {
        return "String: " + s;
    }
}

static void inventarUsuaris(String p) {
    final String[] NOMS = {"Carlos Alberto", "Ana Belén", "Ana Maria", "María", "Victoria Federica", "Julio César", "María Jesús", "Marco Antonio", "Luis Miguel", "Fernando", "Verónica", "Jesús", "Francisco Javier"};
    final String[] COGNOMS = {"Morales", "Caçador", "del Moral", "Díaz", "López Contreras", "Smith", "de la Fuente", "Castro Pérez", "Williams", "Tàpies", "Martí Agell", "Caño", "Fernández Colomer", "García Merino"};

    String[] ids = new String[50];
    String[] nomsCognoms = new String[50];

    int zNdx;
    String id;
    String nom;
    String cognoms;

    for (int i = 0; i < 50; i++) {
        nom = NOMS[((int)(Math.random() * NOMS.length))];
        cognoms = COGNOMS[((int)(Math.random() * COGNOMS.length))];
        zNdx = cognoms.indexOf(" ");
        id = nom.charAt(0) + "." + ((zNdx < 0) ? cognoms : cognoms.substring(0, zNdx));
        id = id.toLowerCase();
        id = eliminarAccents(id);
        int n = ultimId(ids, id, i);
        if (n > 0) id += n;
        ids[i] = id;
        nomsCognoms[i] = cognoms + ", " + nom;
    }

    guardarUsuaris(p, ids, nomsCognoms);
}

static void guardarUsuaris(String p, String[] ids, String[] nomsCognoms) {
    File f = new File(p);
    if (!f.getParentFile().exists()) return;

    try {
        PrintStream writer = new PrintStream(f);

        for (int i = 0; i < ids.length - 1; i++) {
            writer.println(ids[i] + ";" + nomsCognoms[i]);
        }
        writer.print(ids[ids.length - 1] + ";" + nomsCognoms[ids.length - 1]);

        writer.close();
    } catch (IOException e) {
        e.printStackTrace();
    }
}

static int ultimId(String[] ids, String id, int ndx) {
    String s;

    while (ndx > 0) {
        --ndx;
        if (ids[ndx].contains(id)) {
            s = ids[ndx].replace(id, "");    // Es pot fer amb substring(...)
            if (s.length() == 0) {
                return 1;
            } else if (s.matches("^\\d+$")) {  // Es pot fer amb try Integer.parseInt(s)
                return Integer.parseInt(s) + 1;
            }
        }
    }

    return 0;
}

static String eliminarAccents(String s) {
    final String accents = "áàäâéèëêíìïîóòöôúùüûçñ";
    final String sense =   "aaaaeeeeiiiioooouuuucn";
    String ssa = "";
    int ndx;

    for (int i = 0; i < s.length(); i++) {
        ndx = accents.indexOf(s.charAt(i));
        ssa += ((ndx < 0) ? s.charAt(i) : sense.charAt(ndx));
    }

    return ssa;
}

static void notesFinals(String origen, String desti) {
    File forig = new File(origen);
    if (!forig.exists()) return;
    File fdest = new File(desti);
    if (!fdest.getParentFile().exists()) return;

    try {
        Scanner reader = new Scanner(forig);
        PrintStream writer = new PrintStream(fdest);
        String[] camps;

        writer.print(reader.nextLine());
        while (reader.hasNext()) {
            writer.println();
            camps = reader.nextLine().split(";");

            // Calcular mitjanes
            if (Float.parseFloat(camps[2]) < 5 || Float.parseFloat(camps[3]) < 5 || Float.parseFloat(camps[4]) < 5)
                camps[1] = "---";
            else
                camps[1] = String.format("%.1f", (Float.parseFloat(camps[2]) + Float.parseFloat(camps[3]) + Float.parseFloat(camps[4])) / 3);

            if (Float.parseFloat(camps[6]) < 5 || Float.parseFloat(camps[7]) < 5 || Float.parseFloat(camps[8]) < 5)
                camps[5] = "---";
            else
                camps[5] = String.format("%.1f", (Float.parseFloat(camps[6]) + Float.parseFloat(camps[7]) + Float.parseFloat(camps[8])) / 3);

            if (Float.parseFloat(camps[10]) < 5 || Float.parseFloat(camps[11]) < 5 || Float.parseFloat(camps[12]) < 5)
                camps[9] = "---";
            else
                camps[9] = String.format("%.1f", (Float.parseFloat(camps[10]) + Float.parseFloat(camps[11]) + Float.parseFloat(camps[12])) / 3);

            // Guardar
            writer.print(camps[0]); // Nom alumne
            writer.print(";" + camps[1]); // Mitjana M01
            writer.print(";" + camps[2]); // Nota UF1
            writer.print(";" + camps[3]); // Nota UF2
            writer.print(";" + camps[4]); // Nota UF3
            writer.print(";" + camps[5]); // Mitjana M02
            writer.print(";" + camps[6]); // Nota UF1
            writer.print(";" + camps[7]); // Nota UF2
            writer.print(";" + camps[8]); // Nota UF3
            writer.print(";" + camps[9]); // Mitjana M03
            writer.print(";" + camps[10]); // Nota UF1
            writer.print(";" + camps[11]); // Nota UF2
            writer.print(";" + camps[12]); // Nota UF3
        }

        reader.close();
        writer.close();
    } catch (IOException e) {
        e.printStackTrace();
    }
}

static void butlletinsNotes(String p) {
    File forig = new File(p);
    if (!forig.exists()) return;

    try {
        Scanner reader = new Scanner(forig);
        String[] cap = reader.nextLine().split(";");
        String[] dades;

        while (reader.hasNext()) {
            dades = reader.nextLine().split(";");
            butlleti("./Proves/" + dades[0].replaceAll(", ", "") + ".txt", cap, dades);
        }
        reader.close();
    } catch (IOException e) {
        e.printStackTrace();
    }
}

static void butlleti(String dest, String[] cap, String[] dades) {
    File fdest = new File(dest);
    if (!fdest.getParentFile().exists()) return;

    try {
        PrintStream writer = new PrintStream(fdest);

        for (int i = 0; i < cap.length; i++) {
            if (cap[i].charAt(0) == 'U') writer.print("   ");
            writer.println(cap[i] + ": " + dades[i]);
        }

        writer.close();
    } catch (IOException e) {
        e.printStackTrace();
    }
}
