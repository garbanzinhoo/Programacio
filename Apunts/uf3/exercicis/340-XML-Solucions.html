/**
 * Carregar un arxiu XML.
 * @param path Ruta cap a l'arxiu
 * @return Document XML creat a partir de l'arxiu
 */
private static Document loadXmlDocument(String path) {
    DocumentBuilder db;
    Document doc = null;

    try {
        db = dbf.newDocumentBuilder();
        doc = db.parse(new File(path));
    } catch (ParserConfigurationException e) {
        e.printStackTrace();
    } catch (IOException e) {
        e.printStackTrace();
    } catch (SAXException e) {
        e.printStackTrace();
    }

    return doc;
}

/**
 * Mostrar tota la informació dels nodes (i subnodes) d'una llista.
 * @param l Lista de nodes a mostrar
 */
private static void mostrarNodes(LinkedList<Node> l) {
    Node p;

    for (int i = 0; i < l.size(); ++i) {
        p = l.get(i);
        mostrarArbre(p, "");
    }
    System.out.println();
}

/**
 * Mostrar tota la informació d'un node i els seus subnodes.
 * @param n Node a mostrar
 */
private static void mostrarNode(Node n) {
    mostrarArbre(n, "");
    System.out.println();
}

/**
 * Mostrar de forma recursiva tota la informació d'un node i subnodes.<br>
 * Només mostra els nodes de tipus Element i Text.
 * @param n Node del qual es vol mostrar la informació
 * @param tab Tabuladors per indentar cada nivell
 */
private static void mostrarArbre(Node n, String tab) {
    if (n.getNodeType() == Node.TEXT_NODE) {
        // Node tipus Text
        String t = ((Text)n).getWholeText().replaceAll("\\s+", " ").trim();

        // Només mostra el node si no està buit
        if (!t.isBlank()) {
            System.out.print(" --> " + t);
        }
    } else if (n.getNodeType() == Node.ELEMENT_NODE) {
        // Node tipus Element
        Element e = (Element)n;
        String atributs = getAtributs(e);
        System.out.print("\n" + tab + e.getTagName() + (atributs == null ? "" : atributs));

        // Per cada fill, torna a cridar-se a sí mateixa
        Node c = e.getFirstChild();
        while (c != null) {
            mostrarArbre(c, tab + "\t");
            c = c.getNextSibling();
        }
    }
}

/**
 * Obtenir els atributs de l'element 'e'.
 * @param e Element del qual es vol obtenir els atributs
 * @return String amb els atributs de l'element 'e'
 */
private static String getAtributs(Element e) {
    NamedNodeMap atributs = e.getAttributes();
    String s = " (";
    Attr a;
    int i;

    if (atributs.getLength() == 0) return null;

    for (i = 0; i < atributs.getLength() - 1; ++i) {
        a = (Attr)atributs.item(i);
        s += a.getName() + " = " + a.getValue() + ", ";
    }
    a = (Attr)atributs.item(i);
    s += a.getName() + " = " + a.getValue();
    s += ")";

    return s;
}

/**
 * Filtrar i mostrar elements amb un valor concret en un camp concret.
 * @param e Element pare a partir del qual s'han de buscar els elements
 * @param camp Nom del camp
 * @param valor Valor que ha de contenir el camp
 * @param exact 'true' si el valor ha de coïncicir exactament; 'false' si pot ser una subcadena
 * @return Llista amb els nodes que compleixen la condició
 */
private static LinkedList<Node> filtrarCampValor(Element e, String camp, String valor, boolean exact) {
    NodeList l = e.getElementsByTagName(camp);
    LinkedList<Node> le = new LinkedList<>();
    String contingut;
    Node p;

    for (int i = 0; i < l.getLength(); ++i) {
        p = l.item(i);
        contingut = ((Text)p.getFirstChild()).getWholeText().replaceAll("\\s+", " ").trim();
        if (exact) {
            if (contingut.equals(valor)) le.add(p.getParentNode());
        } else {
            if (contingut.contains(valor)) le.add(p.getParentNode());
        }
    }

    return le;
}

/**
 * Filtrar i mostrar elements amb un camp amb el valor dins d'un rang.
 * @param e Element pare a partir del qual s'han de buscar els elements
 * @param preuMin El preu ha de ser superior o igual a aquest valor
 * @param preuMax El preu ha de ser inferior a aquest valor
 * @return Llista amb els nodes que compleixen la condició
 */
private static LinkedList<Node> filtrarMinMax(Element e, String camp, float preuMin, float preuMax) {
    NodeList l = e.getElementsByTagName(camp);
    LinkedList<Node> le = new LinkedList<>();
    Node p;
    String contingut;
    float preu;

    for (int i = 0; i < l.getLength(); ++i) {
        p = l.item(i);
        contingut = ((Text)p.getFirstChild()).getWholeText().replaceAll("\\s+", " ").trim();
        preu = Float.parseFloat(contingut);
        if (preu >= preuMin && preu < preuMax) le.add(p.getParentNode());
    }

    return le;
}

/**
 * Filtrar i mostrar elements amb un valor concret en un camp concret.
 * @param e Element pare a partir del qual s'han de buscar els elements
 * @param camp Nom del camp
 * @param attr Atribut a comparar
 * @param valor Valor que ha de contenir l'atribut
 * @return Llista amb els nodes que compleixen la condició
 */
private static LinkedList<Node> filtrarCampAtributValor(Element e, String camp, String attr, String valor) {
    NodeList l = e.getElementsByTagName(camp);
    LinkedList<Node> le = new LinkedList<>();
    Node p;
    String atribut;

    for (int i = 0; i < l.getLength(); ++i) {
        p = l.item(i);
        atribut = ((Element)p).getAttribute(attr);
        if (atribut.equals(valor)) le.add(p);
    }

    return le;
}

/**
 * Crear un HasMap amb els fills de l'element 'e' utilitzant
 * com a clau un atribut concret.
 * @param e Element del que s'han de guardar els fills en el HasMap
 * @param atribut Atribut dels fills que s'utilitzarà com a clau
 * @return HashMap amb els fills de l'element 'e'
 */
private static HashMap<String,Node> crearHashMap(Element e, String atribut) {
    HashMap<String,Node> l = new HashMap<>();
    Node n = e.getFirstChild();

    while (n != null) {
        if (n.getNodeType() == Node.ELEMENT_NODE) {
            l.put(((Element)n).getAttribute(atribut), n);
        }
        n = n.getNextSibling();
    }

    return l;
}

/**
 * Agrupar i mostrar elements que tenen el mateix valor en un camp concret.<br>
 * Per guardar els nodes agrupats s'utilitza un HashMap en què l'identificador
 * és el preu convertit a Float per poder-lo ordenar fàcilment i el contingut
 * és una llista que contindrà els nodes que tingui el mateix preu.
 * @param e Element pare a partir del qual s'han de buscar els elements
 * @param agruparCamp Nom del camp pel qual s'han d'agrupar
 * @param mostrarCamp Nom del camp que s'ha de mostrar
 */
private static void agruparCampValor(Element e, String agruparCamp, String mostrarCamp) {
    NodeList l = e.getElementsByTagName(agruparCamp);
    HashMap<Float, LinkedList<Node>> grups = new HashMap<>();
    LinkedList<Node> nodes;
    Node p;
    String contingut;
    Float preu;

    for (int i = 0; i < l.getLength(); ++i) {
        p = l.item(i);
        contingut = ((Text)p.getFirstChild()).getWholeText().replaceAll("\\s+", " ").trim();
        preu = Float.parseFloat(contingut);
        // Obtenim la llista corresponent al preu (si existeix)
        nodes = grups.get(preu);
        if (nodes == null) {
            // Si no existeix cap element amb aquest preu,
            // es crea la llista i se li afegeix el node.
            nodes = new LinkedList<>();
            nodes.add(p.getParentNode());
            grups.put(preu, nodes);
        } else {
            // Ja s'ha guardat algun element amb aquest preu
            // per tant només cal afegirlo a la llista
            nodes.add(p.getParentNode());
        }
    }

    // Agafem els preus...
    Set<Float> pr = grups.keySet();
    // ... els fiquem en un array...
    Object[] preus = pr.toArray();
    // ... i ordenem l'array
    Arrays.sort(preus);
    // Per cada element de l'array (per cada preu)...
    for (int i = 0; i < preus.length; ++i) {
        preu = (Float)preus[i];
        System.out.printf("\nPreu: %.2f €", preu);
        // ... obtenim la llista corresponent...
        nodes = grups.get(preu);
        // ... i per cada element...
        for (int j = 0; j < nodes.size(); ++j) {
            // ... obtenim i mostrem el camp que ens interessa
            p = ((Element)nodes.get(j)).getElementsByTagName(mostrarCamp).item(0);
            System.out.print("\n\t" + ((Text)p.getFirstChild()).getWholeText().replaceAll("\\s+", " ").trim());
        }
    }
    System.out.println();
}

/**
 * Augmentar el valor d'un camp de tots els fills de 'e' en un percentatge determinat.
 * @param e Element pare
 * @param camp Camp amb el valor que s'ha d'incrementar
 * @param inc Percentatge en qué s'ha d'incrementar el valor
 */
private static void incrementarCamp(Element e, String camp, float inc) {
    NodeList l = e.getElementsByTagName(camp);
    // Patró per agafar només el número decimal (per si conté altres símbols)
    Pattern pattern = Pattern.compile("(\\d+(\\.\\d+)?)");
    Matcher matcher;
    String contingut;
    Node p;
    float preu;

    for (int i = 0; i < l.getLength(); ++i) {
        p = l.item(i);
        contingut = ((Text)p.getFirstChild()).getWholeText().replaceAll("\\s+", " ").trim();
        matcher = pattern.matcher(contingut);
        if (matcher.find()) {
            // Si troba una part del text que sigui un número decimal
            // agafa només aquesta part
            contingut = matcher.group(1);
            preu = Float.parseFloat(contingut);
            preu += preu * inc / 100;
            ((Text)p.getFirstChild()).replaceWholeText(String.format("%.2f", preu));
        }
    }
}

/**
 * Substitueix el contingut actual de tots els camps d'un tipus
 * amb un valor determinat per un altre valor.<br>
 * @param e Element pare dins del qual s'han de buscar el camps
 * @param camp Nom dels camps que es volen modificar
 * @param vActual Valor que han de tenir actualment
 * @param vNou Valor pel qual s'ha de subtituir
 * @param exact Ha de comprovar que tot el contingut coïncideixi o només una part
 */
private static void canviarCampValor(Element e, String camp, String vActual, String vNou, boolean exact) {
    NodeList l = e.getElementsByTagName(camp);
    String contingut;
    Node p;

    for (int i = 0; i < l.getLength(); ++i) {
        p = l.item(i);
        contingut = ((Text)p.getFirstChild()).getWholeText().replaceAll("\\s+", " ").trim();
        if (exact) {
            if (contingut.equals(vActual)) {
                ((Text)p.getFirstChild()).replaceWholeText(vNou);
            }
        } else {
            if (contingut.contains(vActual)) {
                ((Text)p.getFirstChild()).replaceWholeText(contingut.replaceAll(vActual, vNou));
            }
        }
    }
}

/**
 * Afegir un nou camp (element) amb un valor (text) determinat
 * a tots els fills de l'element 'e'.
 * @param e Element pare al qual s'ha dafegir el nou camp
 * @param camp Nom del camp (element) que s'ha d'afegir
 * @param valor Valor (text) que ha de contenir
 */
private static void afegirFillsCampValor(Element e, String camp, String valor) {
    Document doc = e.getOwnerDocument();
    Node n = e.getFirstChild();
    Node nou;

    while (n != null) {
        if (n.getNodeType() == Node.ELEMENT_NODE) {
            nou = doc.createElement(camp);
            nou.setTextContent(valor);
            n.appendChild(nou);
        }
        n = n.getNextSibling();
    }
}

/**
 * Crea i afegeix un nou element (planta).
 * @param e Element pare on s'ha d'afegir l'element (planta)
 * @param tipus Nom de l'element (PLANT)
 * @param nomComu Nom comú de la planta
 * @param nomBotanic Nom científic de la planta
 * @param zona En quina zona del viver es troba
 * @param llum Quin tipus d'il·luminació necessita
 * @param preu Preu de la planta
 * @param disponible Quantes hi ha de disponibles?
 * @param reg Periodicitat amb què s'ha de regar
 */
private static void afegirElement(Element e, String tipus, String nomComu, String nomBotanic, String zona, String llum, String preu, String disponible, String reg) {
    Document doc = e.getOwnerDocument();
    Node nou;
    Node camp;

    nou = doc.createElement(tipus);
    camp = doc.createElement("COMMON");
    camp.setTextContent(nomComu);
    nou.appendChild(camp);
    camp = doc.createElement("BOTANICAL");
    camp.setTextContent(nomBotanic);
    nou.appendChild(camp);
    camp = doc.createElement("ZONE");
    camp.setTextContent(zona);
    nou.appendChild(camp);
    camp = doc.createElement("LIGHT");
    camp.setTextContent(llum);
    nou.appendChild(camp);
    camp = doc.createElement("PRICE");
    camp.setTextContent(preu);
    nou.appendChild(camp);
    camp = doc.createElement("AVAILABILITY");
    camp.setTextContent(disponible);
    nou.appendChild(camp);
    camp = doc.createElement("WATHERING");
    camp.setTextContent(reg);
    nou.appendChild(camp);

    e.appendChild(nou);
}

/**
 * Crear un document XML amb un element arrel.
 * @param arrel Nom de l'element arrel
 * @return Document XML creat o 'null' si s'ha produït una excepció
 */
private static Document createXmlDocument(String arrel) {
    DocumentBuilder db;
    Document doc = null;
    Node root;

    try {
        db = dbf.newDocumentBuilder();
        doc = db.newDocument();
        root = doc.createElement(arrel);
        doc.appendChild(root);
    } catch (ParserConfigurationException e) {
        e.printStackTrace();
    }

    return doc;
}

/**
 * Guardar un document XML en un arxiu.
 * @param path Ruta cap a l'arxiu on es guardarà el document
 * @param doc Document XML que es vol guardar
 */
private static void saveXmlDocument(String path, Document doc) {
    Transformer t;
    DOMSource source;
    StreamResult result;

    try {
        t = tf.newTransformer();
        source = new DOMSource(doc);
        result = new StreamResult(new File(path));
        t.transform(source, result);
    } catch (TransformerConfigurationException e) {
        e.printStackTrace();
    } catch (TransformerException e) {
        e.printStackTrace();
    }
}
