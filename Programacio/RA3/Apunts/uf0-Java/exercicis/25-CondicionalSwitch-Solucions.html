char op;

System.out.print("Escriu la operació que vols fer (Ex: 2 + 2): ");
n1 = scan.nextDouble();
op = scan.next().charAt(0);
n2 = scan.nextDouble();
scan.nextLine();

// Calcular i mostrar resultat
switch (op) {
    case '+':
        System.out.println("Resultat: " + (n1 + n2));
        break;
    case '-':
        System.out.println("Resultat: " + (n1 - n2));
        break;
    case '*':
        System.out.println("Resultat: " + (n1 * n2));
        break;
    case '/':
        System.out.println("Resultat: " + (n1 / n2));
        break;
    default:
        System.out.println("No reconec l'operació " + op);
}

        // Calcular si és any de traspàs

tos = scan.next();

    System.out.println("Temperatura incorrecta: " + temperatura);
    System.out.println("La temperatura ha d'estar entre 25 i 45");
    return;

tos = tos.toUpperCase();
