final char MIN = 'A', MAX = 'Z';
String frase;
int offset;

System.out.print("Escriu una frase: ");
frase = scan.nextLine();
System.out.print("Quin desplaçament vols? ");
offset = scan.nextInt();

// trim(): elimina espais inicials i finals
// toUpperCase():  converteix a majúscules
// toCharArray(): crea un array de caràcters amb les lletres del String
char[] xifrat = frase.trim().toUpperCase().toCharArray();

// Encriptar
for (int i = 0; i < xifrat.length; i++) {
    // Només si és una lletra entre 'A' i 'Z'
    if (xifrat[i] >= MIN && xifrat[i] <= MAX) {
        xifrat[i] += offset;
        // Si el resultat és més gran que 'Z' agafa una a partir de 'A'
        if (xifrat[i] > MAX) {
            xifrat[i] -= MAX - MIN + 1;
        }
    }
}

// Mostrar array encriptat lletra a lletra
System.out.print("Frase encriptada: ");
for (int i = 0; i < xifrat.length; i++) {
    System.out.print(xifrat[i]);
}
System.out.println();

// Mostrar array encriptat convertint l'array de caràcters a String
String f = new String(xifrat);
System.out.println("Frase encriptada: " + f);

final int MIDA = 10;
String paraula;
String[] paraules = new String[MIDA];

// Introduir paraules
for (int i = 0; i < MIDA; i++) {
    System.out.print("Paraula: ");
    paraules[i] = scan.nextLine();
}

// Ordenar
for (int i = 0; i < MIDA - 1; i++) {
    for (int j = i + 1; j < MIDA; j++) {
        if (paraules[i].compareToIgnoreCase(paraules[j]) > 0) {
            paraula = paraules[i];
            paraules[i] = paraules[j];
            paraules[j] = paraula;
        }
    }
}

// Mostrar paraules ordenades
System.out.println("\nParaules ordenades");
for (int i = 0; i < MIDA; i++) {
    System.out.println(paraules[i]);
}

// Buscar paraules fins que la parauls sigui FI
System.out.println();
do {
    // Demanar paraula
    System.out.print("Paraula: ");
    paraula = scan.nextLine();

    // Buscar paraula
    int min = 0;              // Primera posició de la llista
    int max = MIDA - 1;       // Última posició de la llista
    int mid = 0;              // Posició central (inicialitzar per evitar error)
    int comparacions = 0;

    while (min <= max) {
        // Calcular la posició central del tram entre min i max
        mid = (min + max) / 2;
        ++comparacions;
        if (paraules[mid].compareToIgnoreCase(paraula) < 0) {
            min = mid + 1;        // Agafar la meitat superior
        } else if (paraules[mid].compareToIgnoreCase(paraula) > 0) {
            max = mid - 1;        // Agafar la meitat inferior
        } else {
            break;                // Valor trobat
        }
    }

    // Indicar posició i quantes comparacions ha hagut de fer
    if (MIDA > 0 && paraules[mid].equalsIgnoreCase(paraula)) {
        System.out.println("S'ha trobat la paraula en la posició " + mid +
                           " fent " + comparacions + " comparacions");
    } else {
        System.out.println("NO s'ha trobat la paraula en la llista" +
                           " fent " + comparacions + " comparacions");
    }
} while (!paraula.equalsIgnoreCase("FI"));
