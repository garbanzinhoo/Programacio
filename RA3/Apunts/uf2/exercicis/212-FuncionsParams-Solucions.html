/**
 * Genera un caràcter aleatori entre 'charIni' i 'charFin', tots dos inclosos
 * @param {*} charIni Primer caràcter
 * @param {*} charFin Últim caràcter
 * @returns Un caràcter aleatori entre 'charIni' i 'charFin'
 */
function randomChar(charIni, charFin) {
    // Genera número aleatori entre 65 (A) i 90 (Z)
    let r = Math.floor(Math.random() * (charFin.charCodeAt(0) - charIni.charCodeAt(0) + 1) + charIni.charCodeAt(0));
    // Converteix el número a lletra i la restorna
    return String.fromCharCode(r);
}

/**
 * Genera un caràcter aleatori triat entre els d'un string
 * Es podria millorar passant el string amb els possibles caràcters
 * @returns Un caràcter aleatori
 */
function randomChar2() {
    const LLETRES = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789";
    return LLETRES.charAt(Math.floor(Math.random() * LLETRES.length));
}

/**
 * Crea un array i l'inicialitza amb 'n' elements amb el valor 'v'
 * @param {*} n Nombre d'elements de l'array
 * @param {*} v Valor de cada element
 * @returns Array amb 'n' elements amb el valor 'v'
 */
function initArrayInt(n, v) {
    let a = [];
    for (let i = 0; i < n; ++i) a.push(v);
    return a;
}

/**
 * Converteix l'array a text afegint separadors i/o salts de línia
 * @param {*} a Array
 * @param {*} sep Separador (pcional: per defecte "")
 * @param {*} salt Salt de línia (pcional: per defecte "")
 * @returns Array convertit a text
 */
function arrayToString(a, sep = "", salt = "") {
    return a.join(sep + salt);
}

Solució

/**
 * Converteix un array bidimensional a text afegint separadors i/o salts de línia
 * @param {*} a Array
 * @param {*} sep Separador
 * @param {*} salt Salt de línia
 * @returns Array convertit a text
 */
function arrayBiToString(a, sep = "", salt = "") {
    let r = "";
    for (let fila of a) {
        r += arrayToString(fila, sep) + salt;
    }
    return r;
}

/**
 * Demana un caràcter (opció) a l'usuari i el retorna.
 * Si el caràcter no està entre les opcions, el torna a demanar
 * @param {*} pregunta Pregunta que se li farà a l'usuari
 * @param {*} opcions String amb les possibles opcions
 * @returns El caràcter (opció) triada
 */
function introChar(pregunta, opcions) {
    let r = prompt(pregunta);
    if (r) r = r.trim();
    while (!r || r.length != 1 || !opcions.includes(r)) {
        alert("No és cap d'aquestes opcions: " + opcions);
        r = prompt(pregunta);
        if (r) r = r.trim();
    }
    return r;
}

/**
 * Demana una opció a l'usuari i la retorna.
 * Si l'opció no està entre les opcions possibles, la torna a demanar
 * @param {*} pregunta Pregunta que se li farà a l'usuari
 * @param {*} opcions Array de strings amb les possibles opcions
 * @returns Opció triada
 */
function introChar2(pregunta, opcions) {
    let r = prompt(pregunta);
    if (r) r = r.trim();
    while (!r || !opcions.includes(r)) {
        r = prompt(pregunta);
        if (r) r = r.trim();
    }
    return r;
}

/**
 * Demana un número fins que sigui correcte
 * @param {*} pregunta Pregunta que se li farà a l'usuari
 * @returns Valor introduït
 */
function introInt(pregunta) {
    let r = parseInt(prompt(pregunta));
    while (isNaN(r)) {
        alert("No has introduït cap número")
        r = parseInt(prompt(pregunta));
    }
    return r;
}

/**
 * Demana un número entre 'min' i 'max' fins que sigui correcte
 * @param {*} pregunta Pregunta que se li farà a l'usuari
 * @param {*} min Valor mínim
 * @param {*} max Valor màxim
 * @returns Valor introduït
 */
function introIntMinMax(pregunta, min, max) {
    let r = introInt(pregunta);
    while (r < min || r > max) {
        alert(`El valor ha d'estar entre ${min} i ${max}`);
        r = introInt(pregunta);
    }
    return r;
}

/**
 * Demana 'n' valors entre 'min' i 'max' i els guarda en un array
 * @param {*} pregunta Pregunta que se li farà a l'usuari
 * @param {*} n Nombre de valors que s'han d'introduir
 * @param {*} min Valor mínim
 * @param {*} max Valor màxim
 * @returns Array amb els valors introduïts
 */
function introArrayIntMinMax(pregunta, n, min, max) {
    let a = [];
    for (let i = 0; i < n; ++i) {
        a.push(introIntMinMax(pregunta, min, max));
    }
    return a;
}

/**
 * Inverteix les posicions dels elements de l'array 'a'
 * @param {*} a Array que s'ha d'invertir
 * @returns El mateix array (però amb els elements invetits)
 */
function arrayInvert(a) {
    return a.reverse();
}

/**
 * Inverteix les posicions dels elements de l'array 'a' utilitzat bucle i intercanvi de variables
 * @param {*} a Array que s'ha d'invertir
 * @returns El mateix array (però amb els elements invetits)
 */
function arrayInvert2(a) {
    let t, l = a.length
    for (let i = 0; i < Math.floor(l / 2); ++i) {
        t = a[i];
        a[i] = a[l - i - 1];
        a[l - i - 1] = t;
    }
    return a;   // No necessari
}

/**
 * Desplaça els elements d'un array cap a l'esquerra (el primer al final, el segon al primer...)
 * @param {*} a Array
 */
function arrayToLeft(a) {
    let t = a[0];
    for (let i = 0; i < a.length - 1; ++i) {
        a[i] = a[i + 1];
    }
    a[a.length - 1] = t;
}

Solució desplaçar dreta amb pop i unshift

/**
 * Desplaça els elements d'un array cap a la dreta (l'últim al primer, el primer al segon...)
 * @param {*} a Array
 */
function arrayToRight(a) {
    a.unshift(a.pop());
}

Solució desplaçar dreta o esquerra

/**
 * Desplaça els elements d'un array cap a la dreta o esquerra depenent del paràmetre 'toLeft'
 * @param {*} a Array
 * @param {*} toLeft Si és 'true' desplaça els elements de l'array cap a l'esquerra
 */
function arrayRotate(a, toLeft) {
    if (toLeft) arrayToLeft(a);
    else arrayToRight(a);
}

Solució desplaçar dreta o esquerra N posicions

/**
 * Desplaça els elements d'un array 'n' posicions cap a la dreta o esquerra depenent del paràmetre 'toLeft'
 * @param {*} a Array
 * @param {*} n Nombre de posicions que s'han de desplaçar els elements
 * @param {*} toLeft Si és 'true' desplaça els elements de l'array cap a l'esquerra
 */
function arrayRotateN(a, n, toLeft) {
    for (let i = 0; i < n; ++i) arrayRotate(a, toLeft);
}

/**
 * Barrejar els elements d'un array
 * @param {*} a L'array que s'ha de barrejar
 */
function arrayMix(a) {
    let ndx, t;
    for (let i = 0; i < a.length; ++i) {
        ndx = Math.floor(Math.random() * a.length);
        t = a[i];
        a[i] = a[ndx];
        a[ndx] = t;
    }
}
