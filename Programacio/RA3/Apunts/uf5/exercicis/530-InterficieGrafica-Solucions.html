Solució

En aquesta solució, el Model obté el valor i el modifica directament en la Vista.

Vista

public class Vista extends JFrame {
    private static JTextField comptador;

    // Constructor de la finestra
    public Vista() {
        Container panell = getContentPane();
            // Etiqueta on escriure el valor del comptador
            comptador = new JTextField();
            comptador.setPreferredSize(new Dimension(100, 30));
            comptador.setHorizontalAlignment(SwingConstants.CENTER);
            // Botons per incrementar/decrementar el comptador i controlador
            JButton dButton = new JButton("Dec");
            JButton iButton = new JButton("Inc");
            Controlador ctrl = new Controlador();
            dButton.addActionListener(ctrl);
            iButton.addActionListener(ctrl);
        // Afegir l'etiqueta i el botó a la finestra
        panell.add(comptador, BorderLayout.PAGE_START);
        panell.add(dButton, BorderLayout.LINE_START);
        panell.add(iButton, BorderLayout.LINE_END);

        pack();
        setVisible(true);
        setDefaultCloseOperation(EXIT_ON_CLOSE);
    }

    // Actualitzar el comptador
    public static String getComptador() {
        return comptador.getText();
    }

    // Actualitzar el comptador
    public static void setComptador(String c) {
        comptador.setText(c);
    }

    public static void main(String[] args) {
        new Vista();
        Vista.setComptador("0");
    }
}

Controlador

public class Controlador implements ActionListener {
    @Override
    // Acció quan es clica el botó
    public void actionPerformed(ActionEvent e) {
        // Modificar el comptador
        if (e.getActionCommand().equals("Inc")) Model.incrementar();
        else Model.decrementar();
    }
}

Model

public class Model {
    // Decrementar el valor del comptador i mostrar-lo
    public static void decrementar() {
        int c = Integer.parseInt(Vista.getComptador());
        --c;
        Vista.setComptador("" + c);
    }

    // Incrementar el valor del comptador i mostrar-lo
    public static void incrementar() {
        int c = Integer.parseInt(Vista.getComptador());
        ++c;
        Vista.setComptador("" + c);
    }
}

En aquesta solució, el Controlador fa de pont entre el Model i la Vista.
