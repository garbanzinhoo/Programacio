// Generar lletres entre startChar i endChar, totes dues incloses
static char randomChar(char startChar, char endChar) {
    return (char)(Math.random() * (endChar - startChar + 1) + startChar);
}

static int[] initArrayInt(int nElem, int value) {

    int valor;

// Introduir un número enter entre dos valors
static int introIntMinMax(Scanner scan, int min, int max) {
    int valor;

    if (min > max) {
        int t = min;
        min = max;
        max = t;
    }

    valor = introInt(scan);
    while (valor < min || valor > max) {
        System.out.println("Error: has d'introduir un número entre " + min + " i " + max);
        valor = introInt(scan);
    }

    return valor;
}

// Desplaçar elements array a l'esquerra
static void rotateLeftInt(int[] array) {
    int l = array.length - 1;
    int t = array[0];
    for (int i = 0; i < l; ++i) array[i] = array[i+1];
    array[l] = t;
}

Solució desplaçar dreta

// Desplaçar elements array a la dreta
static void rotateRightInt(int[] array) {
    int l = array.length - 1;
    int t = array[l];
    for (int i = l; i > 0; --i) array[i] = array[i-1];
    array[0] = t;
}

// Desplaçar dreta o esquerra
static void rotateInt(int[] array, char leftRight) {
    if (Character.toUpperCase(leftRight) == 'L') rotateLeftInt(array);
    else rotateRightInt(array);
}

Solució desplaçar dreta o esquerra N posicions

// Desplaçar dreta o esquerra N posicions
static void rotateIntN(int[] array, char leftRight, int nPos) {
    for (int i = 0; i < nPos; ++i) rotateInt(array, leftRight);
}

// Buscar la posició d'un valor enter dins l'array
static int indexOf(int[] array, int value) {
    int min = 0;
    int max = array.length - 1;
    int mid = 0;

    while (min <= max) {
        mid = (min + max) / 2;
        if (array[mid] < value) {
            min = mid + 1;
        } else if (array[mid] > value) {
            max = mid - 1;
        } else {
            break;
        }
    }

    if (array.length > 0 && array[mid] == value) return mid;
    else return -1;
}

// Buscar la posició d'un String dins l'array
static int indexOf(String[] array, String value) {
    int min = 0;
    int max = array.length - 1;
    int mid = 0;

    while (min <= max) {
        mid = (min + max) / 2;
        if (array[mid].compareToIgnoreCase(value) < 0) {
            min = mid + 1;
        } else if (array[mid].compareToIgnoreCase(value) > 0) {
            max = mid - 1;
        } else {
            break;
        }
    }

    if (array.length > 0 && array[mid].equalsIgnoreCase(value)) return mid;
    else return -1;
}
