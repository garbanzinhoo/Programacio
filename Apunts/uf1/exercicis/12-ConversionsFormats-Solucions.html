char lletra = '0';                        // lletra: '0' equival a 48
char lletra2 = 0;                         // lletra2: 0
boolean sonIguals = lletra == lletra2;    // sonIguals: false
lletra = 'A';                             // lletra: 'A' equival a 65
lletra2 = 65;                             // lletra2: 65
sonIguals = lletra == lletra2;            // sonIguals: true
++lletra;                                 // lletra: 'B' equival a 66
lletra = '0' + '1';                       // lletra: 'a' equival a 97 (48 + 49)

System.out.printf("|%20d|\n", 53);             // |                  53|
System.out.printf("|%-20d|\n", 53);            // |53                  |
System.out.printf("|%020d|\n", 53);            // |00000000000000000053|
System.out.printf("%,d\n", 10000000);          // 10.000.000
System.out.printf("%.2f\n", 12.9f);            // 12,90
System.out.printf("%,.2f\n", 54999.99789f);    // 55.000,00
