Mostrar informació detallada

static void showDetailedInfo(File f) {
    try {
        System.out.println("\nCanonical path: " + f.getCanonicalPath());
    } catch (IOException e) {
        System.out.println("\nError al obtenir la ruta canònica:");
        e.printStackTrace();
        return;
    }
    System.out.println("Directory:      " + f.isDirectory());
    System.out.println("Hidden:         " + f.isHidden());
    System.out.println("Read:           " + f.canRead());
    System.out.println("Write:          " + f.canWrite());
    System.out.println("Execute:        " + f.canExecute());
    System.out.println("Length:         " + f.length());
    Date d = new Date(f.lastModified());
    System.out.println("LastModified:   " + d.toString());
}

static void showDetailedInfo(String p) {
    File f = new File(p);

    if (f.exists()) showDetailedInfo(f);
    else System.out.println("\nNo s'ha trobat l'arxiu o carpeta " + p);
}

static String getAttributes(File f) {
    String attr = "";

    attr += f.isDirectory() ? "D" : "-";
    attr += f.isHidden() ? "H" : "-";
    attr += f.canRead() ? "R" : "-";
    attr += f.canWrite() ? "W" : "-";
    attr += f.canExecute() ? "X" : "-";

    return attr;
}

static String getDateSize(File f) {
    Calendar c = Calendar.getInstance();
    c.setTimeInMillis(f.lastModified());

    int y = c.get(Calendar.YEAR);
    int m = c.get(Calendar.MONTH) + 1;
    int d = c.get(Calendar.DAY_OF_MONTH);
    int hh = c.get(Calendar.HOUR);
    int mm = c.get(Calendar.MINUTE);

    return String.format("%02d/%02d/%04d %02d/%02d %12d", d, m, y, hh, mm, f.length());
}

static String convertSize(long b) {
    final String[] B = {"  B", " KB", " MB", " GB", " TB"};
    int i = 0;

    while (!(b / 1000 == 0 || i == B.length - 1)) {
        b /= 1000;
        ++i;
    }

    return b + B[i];
}

Obtenir informació dels sistemes d'arxius

static String[] sistemesFitxers() {
    File[] roots = File.listRoots();
    String[] sf = new String[roots.length];

    for (int i = 0; i < roots.length; i++) {
        sf[i] = roots[i].toString();
    }

    return sf;
}

static long[] midaEspaiLliure(String sf) {
    File f = new File(sf);
    long[] mides = new long[2];

    mides[0] = f.getTotalSpace();
    mides[1] = f.getUsableSpace();
    //mides[1] = f.getFreeSpace();

    return mides;
}

static void showInfoFS() {
    String[] sf = sistemesFitxers();
    String mides;
    long[] m;

    System.out.println("\nSistemes de fitxers (v1)");
    for (String s: sf) {
        m = midaEspaiLliure(s);
        mides = String.format("%8s %8s", convertSize(m[0]), convertSize(m[0]));
        System.out.format("%s   %s\n", mides, s);
    }
}

// AVANÇAT
static void showInfoFS2() {
    long total, avail;
    String mides, s;

    System.out.println("\nSistemes de fitxers (v2)");
    for (FileStore store : FileSystems.getDefault().getFileStores()) {
        try {
            total = store.getTotalSpace();
            avail = store.getUsableSpace();
            //avail = store.getUnallocatedSpace();
            mides = String.format("%8s %8s", convertSize(total), convertSize(avail));
            s = store.toString();
            if (avail > 0 && (s.contains("(/dev") || s.contains(":)"))) {
                System.out.printf("%s   %s\n", mides, s);
            }
        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}

static void dir(String p) {
    File f = new File(p);
    if (!f.exists() || !f.isDirectory()) {
        System.out.println("\nNo existeix la ruta o no és una carpeta");
        return;
    }

    File[] e = f.listFiles();
    int n = e.length;
    int na = 0, nc = 0;
    for (int i = 0; i < n; i++) {
        System.out.println(getFileInfo(e[i]));
        if (e[i].isDirectory()) ++nc;
        else ++na;
    }
    System.out.println("      " + na + " arxius");
    System.out.println("      " + nc + " carpetes");
}

static void crearEstructura() {
    // Tornar a crear l'estructura de proves
    File f = new File("./Proves/Subcarpeta");
    f.mkdirs();
    f = new File("./Proves/arxiu.txt");
    try {
        f.createNewFile();
    } catch (IOException e) {
        System.out.println("\nNo s'ha pogut crear l'arxiu " + f);
    }

    // Afegir carpeta i arxiu
    f = new File("./Proves/Proves2");
    f.mkdir();
    f = new File("./Proves/Proves2/arxiu2.txt");
    try {
        f.createNewFile();
    } catch (IOException e) {
        System.out.println("\nNo s'ha pogut crear l'arxiu " + f);
    }

    // Renombrar arxiu i esborrar arxiu i carpeta
    File fNou = new File("./Proves/Proves2/arxiu3.txt");
    f.renameTo(fNou);
    fNou.delete();
    f = new File("./Proves/Proves2");
    f.delete();
}
