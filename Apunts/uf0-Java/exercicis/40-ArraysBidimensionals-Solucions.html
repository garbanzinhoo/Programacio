int i, j, t, fc, suma;
boolean filesIguals, columnesIguals, diagonalsIguals, numerosTots;

// Introduir mida
//System.out.print("Introdueix la mida de la matriu: ");
fc = scan.nextInt();

while (fc != 0) {
    // Crear i llegir array
    int[][] array = new int[fc][fc];
    for (i = 0; i < fc; ++i) {
        for (j = 0; j < fc; ++j) {
            array[i][j] = scan.nextInt();
        }
    }

    // Comprovar files
    suma = -1;
    filesIguals = true;
    for (i = 0; i < fc; ++i) {
        t = 0;
        for (j = 0; j < fc; ++j) {
            t += array[i][j];
        }
        if (suma == -1) suma = t;
        if (suma != t) {
            filesIguals = false;
            break;
        }
    }

    // Comprovar columnes
    suma = -1;
    columnesIguals = true;
    for (i = 0; i < fc; ++i) {
        t = 0;
        for (j = 0; j < fc; ++j) {
            t += array[j][i];
        }
        if (suma == -1) suma = t;
        if (suma != t) {
            columnesIguals = false;
            break;
        }
    }

    // Comprovar diagonals
    diagonalsIguals = true;
    t = 0;
    for (i = 0; i < fc; ++i) {
        t += array[i][i];
    }
    suma = t;
    t = 0;
    for (i = 0; i < fc; ++i) {
        t += array[i][fc - i - 1];
    }
    if (suma != t) {
        diagonalsIguals = false;
    }

    // Comprovar si hi son tots els  números
    numerosTots = true;
    boolean[] numeros = new boolean[fc * fc + 1];
    for (i = 0; i < fc; ++i) {
        for (j = 0; j < fc; ++j) {
            t = array[i][j];
            if (t >= 1 && t <= fc * fc) {
                if (numeros[t]) {
                    numerosTots = false;
                    break;
                } else {
                    numeros[t] = true;
                }
            } else {
                numerosTots = false;
                break;
            }
        }
        if (!numerosTots) break;
    }

    // Mostrar resultat
    if (filesIguals && columnesIguals && diagonalsIguals && numerosTots) {
        System.out.println("És un quadrat màgic");
    } else {
        System.out.println("No és un quadrat màgic:");
        if (!filesIguals) System.out.println("   -No totes les files sumen el mateix");
        if (!columnesIguals) System.out.println("   -No totes les columnes sumen el mateix");
        if (!diagonalsIguals) System.out.println("   -Les dues diagonals no sumen el mateix");
        if (!numerosTots) System.out.println("   -No conté tots els números");
    }

    // Introduir mida
    //System.out.print("\nIntrodueix la mida de la matriu: ");
    fc = scan.nextInt();
}
