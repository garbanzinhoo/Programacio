double n2;

System.out.print("Introdueix el primer valor: ");
n1 = scan.nextDouble();
scan.nextLine();
System.out.print("Introdueix el segon valor: ");
n2 = scan.nextDouble();
scan.nextLine();
System.out.print("Quina operació vols fer (+, -, * o /)? ");
operacio = scan.nextLine().charAt(0);
 
if (operacio == '+') System.out.println("El resultat és " + (n1 + n2));
else if (operacio == '-') System.out.println("El resultat és " + (n1 - n2));
else if (operacio == '*') System.out.println("El resultat és " + (n1 * n2));
else if (operacio == '/') System.out.println("El resultat és " + (n1 / n2));
else System.out.println("Error: operació no reconeguda");
