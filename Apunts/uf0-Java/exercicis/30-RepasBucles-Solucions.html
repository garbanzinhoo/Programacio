String s = "-123.456";
boolean esNegatiu = false;
boolean esReal = false;
boolean esZero = false;
char c;
int i = 0;
int ndx = -1;
int punts = 0;
int l = s.length();

// Comprovar si té signe, i si és negatiu
c = s.charAt(i);
if (c == '-') {
    esNegatiu = true;
    ++i;
} else if(c == '+') {
    ++i;
}

// Comprovar si té algun caràcter que no sigui 0-9 i que no tingui més d'un punt
// També comprova si és real (si té punt i algun dels dígits no és 0)
for ( ; i < l; ++i) {
    c = s.charAt(i);
    if (c == '.') {
        ndx = i;    // Guarda la posició del punt
        ++punts;
        if (punts > 1) return;
    } else {
        if (c < '0' || c > '9') return;
        if (c != '0' && ndx >= 0) esReal = true;
    }
}

// Si no és real, comprova si tot a l'esquerra del punt són 0s
if (!esReal) {
    esZero = true;
    if (ndx >= 0) i = ndx;
    else i = l;
    do {
        --i;
        c = s.charAt(i);
        if (c != '0' && c != '+' && c != '-') esZero = false;
    } while (i > 0);
}

// Separar i mostrar part entera i decimals
System.out.print("Part entera: ");
if (!esReal) ndx = l;
for (i = 0; i < ndx; ++i) System.out.print(s.charAt(i));
System.out.println();
if (esReal) {
    System.out.print("Part decimal: ");
    for (i = ndx + 1; i < l; ++i) System.out.print(s.charAt(i));
    System.out.println();
}

System.out.println(s + " és un número " + (esReal ? "real" : "enter"));
if (esZero) System.out.println(s + " té el valor 0");
else System.out.println(s + " és " + (esNegatiu ? "negatiu" : "positiu"));
