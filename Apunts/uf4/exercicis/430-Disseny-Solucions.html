public class Zoo {
    ArrayList<Animal> animals = new ArrayList<>();

    public ArrayList getAnimals() {
        return animals;
    }

    public void afegirAnimal(Animal animal) {
        afegirAnimal(animal, 0);
    }

    public void afegirAnimal(Animal animal, int quants) {
        if (animal.getGrup() == null || animal.getAlimentacio() == null) return;

        Animal a = buscarAnimal(animal.getNom());

        if (a != null) {
            a.afegir(quants);
        } else {
            animal.afegir(quants);
            animals.add(animal);
        }
    }

    public void afegirAnimal(String animal, int quants) {
        Animal a = buscarAnimal(animal);

        if (a != null) {
            a.afegir(quants);
        }
    }

    public Animal buscarAnimal(String nom) {
        for (Animal animal: animals) {
            if (animal.getNom().equals(nom)) return animal;
        }
        return null;
    }

    public int quantsTipus() {
        return animals.size();
    }

    public int quantsAnimals() {
        int n = 0;
        for (Animal animal: animals) {
            n += animal.getQuantitat();
        }
        return n;
    }

    public int quantsAnimalsGrup(String grup) {
        int n = 0;
        for (Animal animal: animals) {
            if (animal.getGrup().equals(grup)) {
                n += animal.getQuantitat();
            }
        }
        return n;
    }

    public int quantsAnimalsAlimentacio(String alimentacio) {
        int n = 0;
        for (Animal animal: animals) {
            if (animal.getAlimentacio().equals(alimentacio)) {
                n += animal.getQuantitat();
            }
        }
        return n;
    }

    public ArrayList filtrarGrup(String grup) {
        ArrayList<Animal> a = new ArrayList<>();
        for (Animal animal: animals) {
            if (animal.getGrup().equals(grup)) {
                a.add(animal);
            }
        }
        return a;
    }

    public ArrayList filtrarAlimentacio(String alimentacio) {
        ArrayList<Animal> a = new ArrayList<>();
        for (Animal animal: animals) {
            if (animal.getAlimentacio().equals(alimentacio)) {
                a.add(animal);
            }
        }
        return a;
    }
}

Animal

public class Animal {
    static final String[] GRUPS = {"Mamífer", "Peix", "Au", "Rèptil"};
    static final String[] ALIMENTACIO = {"Herbívor", "Carnívor", "Omnívor"};

    String nom;
    String grup;
    String alimentacio;
    int quantitat = 0;

    public Animal(String _nom, String _grup, String _alimentacio) {
        nom = _nom;
        if (!Arrays.asList(GRUPS).contains(_grup)) grup = null;
        else grup = _grup;
        if (!Arrays.asList(ALIMENTACIO).contains(_alimentacio)) alimentacio = null;
        else alimentacio = _alimentacio;
    }

    public void afegir(int quants) {
        if (quants > 0) quantitat += quants;
    }

    public void treure(int quants) {
        if (quants > 0 && quantitat >= quants) quantitat -= quants;
    }

    public String getNom() {
        return nom;
    }

    public String getGrup() {
        return grup;
    }

    public String getAlimentacio() {
        return alimentacio;
    }

    public int getQuantitat() {
        return quantitat;
    }

    public String toString() {
        return quantitat + " " + nom + " " + grup + " " + alimentacio;
    }
}
