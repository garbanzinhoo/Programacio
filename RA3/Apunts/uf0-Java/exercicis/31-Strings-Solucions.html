String s = "   Administració   de   sistemes   informàtics   en   xarxa   ";
char cActual, cAnterior;

// Eliminar possibles espais inicials i/o finals i convertir a majúscules
// Afegeix un espai al final per simplificar les següents comprovacions
s = s.trim().toUpperCase() + " ";
// Escriure la primera lletra si la paraula té més de 3 lletres
if (s.indexOf(" ") > 3) System.out.print(s.charAt(0));
// A partir del segon caràcter...
for (int i = 1; i < s.length(); i++) {
    cActual = s.charAt(i);
    cAnterior = s.charAt(i - 1);
    // ... escriure'l si no és un espai i l'anterior sí...
    // Aquest mètode funciona encara que hi hagi més d'un espai entre les paraules
    if (cActual != ' ' && cAnterior == ' ') {
        // ... i si el següent espai està almenys 4 posicions més lluny
        if (s.indexOf(" ", i) > (i + 3)) System.out.print(cActual);
    }
}
System.out.println();

Solució substituir múltiples espais

String s = "   Hi   havia   una   vegada...   ";
String s2 = "";

s = s.trim();
for (int i = 0; i < s.length(); i++) {
    if (s.charAt(i) == ' ') {
        // Avança 'i' mentre segueixi trobant espais
        do {
            ++i;
        } while (s.charAt(i) == ' ');
        s2 += '|';
    }
    s2 += s.charAt(i);
}

System.out.println(s2);

String s = "   Administració   de   sistemes   informàtics   en   xarxa   ";
String p1;
int ndx;

s = s.trim();            // Elimina espais al principi i al final
ndx = s.indexOf(" ");    // Busca la posició del primer espai
while (ndx >= 0) {
    // Agafa la primera paraula i l'escriu
    p1 = s.substring(0, ndx);
    System.out.println(p1);

    // Agafa la resta de la frase i elimina espais al principi
    s = s.substring(ndx).trim();
    ndx = s.indexOf(" ");
}

String s = "MuRcIeLaGo";
String s2 = "";
char c;
int ndx;

System.out.println(s);

// Mentre quedin lletres al paraula
while (s.length() > 0) {
    // De moment, la més petita és la primera
    c = s.charAt(0);
    ndx = 0;
    // Comparar la més petita amb les altres...
    for (int i = 1; i < s.length(); ++i) {
        // ... si troba una més petita, la guarda i indica la posició
        if (s.charAt(i) < c) {
            c = s.charAt(i);
            ndx = i;
        }
    }
    s2 += c;    // Afegeix la més petita a la llista de lletres ordenades
    // Elimina la lletra de la paraula original:
    //  concatena les que hi ha abans amb les que hi ha després
    s = s.substring(0, ndx) + s.substring(ndx + 1);
}

System.out.println(s2);
