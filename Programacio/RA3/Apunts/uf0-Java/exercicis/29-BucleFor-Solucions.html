Una forma més ràpida i que permet especificar el nombre de decimals (definit en la variable nDigits):

// Solució per calcular N decimals utilitzant el mètode de l’arc tangent
import java.math.BigInteger;
import java.time.Instant;
import java.time.Duration;
 
public class Main {
    public static BigInteger ArctanDenom(BigInteger d, int nDigits) {
        BigInteger total = (BigInteger.TEN.pow(nDigits)).divide(d);
        BigInteger term = total;
        BigInteger n = BigInteger.ZERO;
        while (term.compareTo(BigInteger.ZERO) != 0) {
            n = n.add(BigInteger.ONE);
            term = term.divide(d.multiply(d.negate()));
            total = total.add(term.divide(n.add(n).add(BigInteger.ONE)));
        }
        System.out.printf("ArctanDenom(%d) took %d iterations\n", d, n);
        return total;
    }
 
    public static void main(final String[] args) {
        String resultat;
        int xDigits = 10;
        int nDigits = 10000;
 
        BigInteger a4 = new BigInteger("4");
        BigInteger a5 = new BigInteger("5");
        BigInteger a239 = new BigInteger("239");
 
        BigInteger pi = a4.multiply((a4.multiply(ArctanDenom(a5, nDigits + xDigits))).subtract(ArctanDenom(a239, nDigits + xDigits)));
        Instant ini = Instant.now();
        pi = pi.divide(BigInteger.TEN.pow(xDigits));
        resultat = pi.toString();
        Instant fi = Instant.now();
        Duration temps = Duration.between(ini, fi);
 
        System.out.println(resultat);
        System.out.println("Temps: " + temps.toMillis() + "ms");
        System.out.println("Decimals: " + (resultat.length() - 1));
    }
}
