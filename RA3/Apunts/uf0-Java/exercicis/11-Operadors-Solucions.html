System.out.println("Operador AND (&&)");
System.out.println("false && false --> " + (false && false));
System.out.println("false && true --> " + (false && true));
System.out.println("true && false --> " + (true && false));
System.out.println("true && true --> " + (true && true));
System.out.println();
System.out.println("Operador OR (||)");
System.out.println("false || false --> " + (false || false));
System.out.println("false || true --> " + (false || true));
System.out.println("true || false --> " + (true || false));
System.out.println("true || true --> " + (true || true));
System.out.println();
System.out.println("Operador NOT (!)");
System.out.println("!false --> " + (!false));
System.out.println("!true --> " + (!true));

Operador AND (&&)
false && false --> false
false && true --> false
true && false --> false
true && true --> true

Operador OR (||)
false || false --> false
false || true --> true
true || false --> true
true || true --> true

Operador NOT (!)
!false --> true
!true --> false

final int IVA = 21;
float preu = 123.45f;
float preuDTE = preu - preu * 15 / 100;              // DTE (15) és un literal
// float preuDTE = preu * (1 - (float)15 / 100);      // Si no es converteix, ERROR!
float preuIVA = preuDTE + preuDTE * IVA / 100;        // IVA és una constant
// float preuIVA = preuDTE * (1 + (float)IVA / 100);  // Si no es converteix, ERROR!
System.out.println(preu);
System.out.println(preuDTE);
System.out.println(preuIVA);
