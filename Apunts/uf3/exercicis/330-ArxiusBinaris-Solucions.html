Les funcions guardarText() i llegirText() s'utilitzen en els següents exercicis per guardar i llegir camps de text en un arxiu binari:

public static void guardarText(RandomAccessFile raf, long pos, String text, int maxSize) throws IOException{
    byte[] b = new byte[maxSize];
    byte[] bs = text.getBytes(StandardCharsets.UTF_8);

    if (bs.length > b.length - 1) {
        System.out.println("Error: text massa llarg");
        // throw new Exception("Error: text massa llarg");
        return;
    }

    System.arraycopy(bs, 0, b, 0, bs.length);
    if (pos >= 0) raf.seek(pos);
    raf.write(b);
}

public static String llegirText(RandomAccessFile raf, long pos, int maxSize) throws IOException{
    byte[] b = new byte[maxSize];
    if (pos >= 0) raf.seek(pos);
    raf.read(b);
    String s = new String(b, StandardCharsets.UTF_8);
    int z = s.indexOf((char)0);
    if (z >= 0) s = s.substring(0, z);
    return s;
}

static void llegirArxiuBinariHexadecimal(String p) {
    final int N_BYTES = 20;
    File f = new File(p);
    if (!f.exists()) return;

    try {
        RandomAccessFile raf = new RandomAccessFile(f, "r");
        long l = raf.length();
        byte[] b = new byte[N_BYTES];
        long pos = 0;

        while (l > 0) {
            // Llegir línia
            raf.read(b);

            // Mostrar posició
            System.out.printf("%08d  ", pos);

            // Mostrar bytes en hexadecimal
            for (int i = 0; i < N_BYTES; i++) {
                if (i < l) {
                    System.out.print(String.format("%02X ", b[i]));
                } else {
                    System.out.print("   ");
                }
            }
            System.out.print(" ");

            // Mostrar bytes en ASCII
            for (int i = 0; i < N_BYTES; i++) {
                if (i < l) {
                    if (b[i] < 32 || b[i] >= 127) System.out.print("\u001b[31m.\u001b[0m");
                    else System.out.print((char)(b[i] & 0xFF));
                } else {
                    System.out.print(" ");
                }
            }
            System.out.println();

            pos += N_BYTES;
            l -= N_BYTES;
        }
    } catch (Exception e) {
        e.printStackTrace();
    }
}

static void guardarNumerosBinaris(String p) {
    File f = new File(p);
    if (!f.getParentFile().exists()) return;

    try {
        RandomAccessFile raf = new RandomAccessFile(f, "rw");

        for (int i = 0; i < 10; i++) raf.writeByte((byte)i);
        for (int i = 0; i < 10; i++) raf.writeShort((short)i);
        for (int i = 0; i < 10; i++) raf.writeInt(i);
        for (int i = 0; i < 10; i++) raf.writeLong(i);
        for (int i = 0; i < 10; i++) raf.writeFloat(i);
        for (int i = 0; i < 10; i++) raf.writeDouble(i);
    } catch (Exception e) {
        e.printStackTrace();
    }
}

static void mostrarPosicioTipus(RandomAccessFile raf, long pos, char type) throws IOException {
    switch (type) {
        case 'a':
        case 'b':
            raf.seek(pos);
            System.out.println("Byte posicio " + pos + ":    " + raf.readByte());
            if (type != 'a') break;
        case 's':
            raf.seek(pos);
            System.out.println("Short posicio " + pos + ":   " + raf.readShort());
            if (type != 'a') break;
        case 'i':
            raf.seek(pos);
            System.out.println("Integer posicio " + pos + ": " + raf.readInt());
            if (type != 'a') break;
        case 'l':
            raf.seek(pos);
            System.out.println("Long posicio " + pos + ":    " + raf.readLong());
            if (type != 'a') break;
        case 'f':
            raf.seek(pos);
            System.out.println("Float posicio " + pos + ":   " + raf.readFloat());
            if (type != 'a') break;
        case 'd':
            raf.seek(pos);
            System.out.println("Double posicio " + pos + ":  " + raf.readDouble());
            break;
        default:
            System.out.println("Tipus desconegut");
    }
}

static void guardarRegistresText(String p) {
    final int TEXT_SIZE = 40;
    final int MIDA_REGISTRE = TEXT_SIZE + Float.BYTES + Integer.BYTES;

    File f = new File(p);
    if (!f.getParentFile().exists()) return;

    String[] pacients = {"Pau Gasoil", "Mireia del Monte", "Andrés Lasiesta"};
    float[] altures = {1.98f, 1.76f, 1.71f};
    int[] pesos = {93, 71, 75};

    try {
        RandomAccessFile raf = new RandomAccessFile(f, "rw");

        for (int i = 0; i < pacients.length; i++) {
            modificarRegistre(raf, i, MIDA_REGISTRE, pacients[i], altures[i], pesos[i]);
        }
    } catch (Exception e) {
        e.printStackTrace();
    }
}

static void llegirRegistresText(String p) {
    final int TEXT_SIZE = 40;
    final int MIDA_REGISTRE = TEXT_SIZE + Float.BYTES + Integer.BYTES;

    File f = new File(p);
    if (!f.getParentFile().exists()) return;

    String pacient;
    float altura;
    int pes;
    long mida;
    long regs;

    try {
        RandomAccessFile raf = new RandomAccessFile(f, "rw");
        mida = raf.length();
        regs = mida / MIDA_REGISTRE;

        for (int i = 0; i < regs; i++) {
            pacient = llegirText(raf, -1, TEXT_SIZE);
            altura = raf.readFloat();
            pes = raf.readInt();
            System.out.printf("%-20s %4.2f m   %3d Kg\n", pacient, altura, pes);
        }
    } catch (Exception e) {
        e.printStackTrace();
    }
}

static void modificarRegistresText(String p) {
    final int TEXT_SIZE = 40;
    final int MIDA_REGISTRE = TEXT_SIZE + Float.BYTES + Integer.BYTES;

    File f = new File(p);
    if (!f.getParentFile().exists()) return;

    try {
        RandomAccessFile raf = new RandomAccessFile(f, "rw");

        modificarRegistre(raf, 0, MIDA_REGISTRE, "Peter Punk", 1.82f, 79);
        modificarRegistre(raf, 1, MIDA_REGISTRE, "Mireia del Monte", 1.74f, 71);
        modificarRegistre(raf, 2, MIDA_REGISTRE, "Andrés Lasiesta", 1.71f, 77);
    } catch (Exception e) {
        e.printStackTrace();
    }
}

static void modificarRegistre(RandomAccessFile raf, long reg, int mida, String pacient, float altura, int pes) throws IOException {
    final int TEXT_SIZE = 40;

    raf.seek(reg * mida);
    guardarText(raf, -1, pacient, TEXT_SIZE);
    raf.writeFloat(altura);
    raf.writeInt(pes);
}
